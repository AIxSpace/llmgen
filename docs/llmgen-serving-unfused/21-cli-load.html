阶段 6 · 收尾

CLI 与快速加载

正式的 chat / bench 子命令；权重不经过进程堆。

本步目标把一路上攒下的环境变量开关整理成正式的命令行（chat 和 bench 两个子命令），bench 输出和 lab、nano-vllm 同格式的 JSON。顺手把加载改成“权重不经过进程堆”，RSS 降 1.5 GB，加载到 1 秒以内。

原理

为什么要正式的命令行

到现在为止的开关：LLMGEN_MODEL、LLMGEN_WORKLOAD、LLMGEN_EAGER、LLMGEN_SYNC_EACH、LLMGEN_POLICY、LLMGEN_PROFILE、LLMGEN_METRICS……prompt 和 max_model_len 还写死在 llmgen.zig 里。下一步要用脚本批量跑四个场景、和另外两个引擎对比，需要：

  • 每次运行的参数能从命令行完整给出、能记在日志里；
  • 结果是机器可读的 JSON，字段和另外两个引擎一致，一个 report.py 就能汇总。

bench 的口径

和 lab、bench_nanovllm.py 完全相同，这是三方数字能放在一张表里的前提：

口径
预热正式计时前跑一条 8 token 输入、64 token 输出的短请求。吃掉 GPU 升频、cuBLAS 初始化、第一次 graph launch
提交一轮的所有请求在 t0 同时 addRequest
TTFT每条请求第一个 token 回到 host 的时刻 − t0（含排队）
ITL同一请求相邻两个 token 的间隔，所有请求的样本合在一起算 mean / p50 / p99
吞吐一轮的输出 token 数 ÷ 一轮的墙钟。多轮取墙钟的中位数
峰值 RSS/proc/self/status 的 VmHWM
峰值显存进程外用 NVML 每 50 ms 采样（gpu_mon.py），不信任引擎自报

加载：权重为什么不该经过进程堆

src/gguf.zig · GGUFParser.init 起点（你现在的代码）
    pub fn init(allocator: std.mem.Allocator, filename: []const u8) !Self {
        const file = try std.fs.cwd().openFile(filename, .{});
        defer file.close();

        const file_stat = try file.stat();
        const file_size = file_stat.size;
        const buffer = try allocator.alloc(u8, @intCast(file_size));
        errdefer allocator.free(buffer);

        var offset: usize = 0;
        while (offset < buffer.len) {
            const n = try file.read(buffer[offset..]);
            if (n == 0) break;
            offset += n;
        }

        const gguf_info = try parse(allocator, buffer); // 开始解析 gguf 格式

        return .{
            .gguf_info = gguf_info,
            .buffer = buffer,
            .data_offset = gguf_info.data_offset,
            .allocator = allocator,
        };
    }

起点把整个 GGUF 文件（1.5 GB）读进一块堆内存，解析头部，然后逐个张量从这块内存 H2D。两个问题：

  • RSS：这 1.5 GB 一直占着，直到 parser 析构。目标 lab 的峰值 RSS 是 519 MB。
  • 拷贝：从普通（pageable）内存 H2D，驱动内部要先拷到它自己的 pinned 暂存区，等于多拷一遍。

lab 的做法：只读文件开头的一小段（16 MB，不够就翻倍）来解析头部；权重数据留在磁盘上，加载时 pread 直接读进一块 32 MB 的 pinned 暂存区，再 H2D。

gguf_parser.zig lab 参考（3090 实测过）
    /// 对照 GGUFParser::GGUFParser(filename).
    /// [serving] 与 module-style 的差异:只读文件**前缀**直到头部能完整解析(不够就翻倍重读),
    /// 张量数据留在磁盘,由 readTensor 按需 pread —— 1.5GB 的 BF16 权重不经过进程堆.
    pub fn init(alloc: std.mem.Allocator, filename: []const u8) !GGUFParser {
        const file = try std.fs.cwd().openFile(filename, .{});
        errdefer file.close();
        const size: usize = @intCast((try file.stat()).size);
        var want: usize = @min(size, 16 << 20);
        while (true) {
            const buffer = try alloc.alloc(u8, want);
            const n = try file.preadAll(buffer, 0);
            if (parse(alloc, buffer[0..n])) |parsed| {
                return .{
                    .info_ = parsed.info,
                    .buffer = buffer,
                    .data_offset_ = parsed.info.offset,
                    .file = file,
                    .alloc = alloc,
                };
            } else |e| {
                alloc.free(buffer);
                if (e != ParseError.OutOfBounds or want >= size) return e;
                want = @min(size, want * 2);
            }
        }
    }

    /// 把某张量的原始磁盘字节(bf16/f16/f32 未反量化)读进 dst(长度须 == t.bytes()).
    pub fn readTensor(self: *const GGUFParser, t: *const TensorInfo, dst: []u8) !void {
        std.debug.assert(dst.len == t.bytes());
        const n = try self.file.?.preadAll(dst, self.data_offset_ + t.offset);
        if (n != dst.len) return error.ShortRead;
    }

            else if (std.mem.eql(u8, key, "max-batched-tokens")) o.max_batched = try std.fmt.parseInt(usize, v, 10) //
            else if (std.mem.eql(u8, key, "max-model-len")) o.max_model_len = try std.fmt.parseInt(usize, v, 10) //

            else if (std.mem.eql(u8, key, "kv-blocks")) o.kv_blocks = try std.fmt.parseInt(usize, v, 10) //
            else if (std.mem.eql(u8, key, "policy")) o.policy = v //
            else if (std.mem.eql(u8, key, "chunk")) o.chunk = try std.fmt.parseInt(usize, v, 10) //
            else if (std.mem.eql(u8, key, "pass-tokens")) o.pass_tokens = try std.fmt.parseInt(usize, v, 10) //

fn chat(alloc: std.mem.Allocator, w: *std.Io.Writer, parser: anytype, engine: anytype, o: Options) !void {
    const tkz = @import("tokenizer.zig");
    var tok = try tkz.Tokenizer.fromGguf(alloc, parser);
    defer tok.deinit();
    const text = o.arg0 orelse "中国的首都是哪里？";
    const templated = try tkz.qwenChatTemplate(alloc, text, true);
    defer alloc.free(templated);
    const ids = try tok.encode(templated);
    defer alloc.free(ids);
    try w.print("prompt: \"{s}\" ({d} tokens)\n", .{ text, ids.len });
    try w.flush();

    const seq = try engine.addRequest(ids, .{ .temperature = o.temp, .max_tokens = o.max_new, .ignore_eos = false });
    const t0 = std.time.nanoTimestamp();
    var t_first: i128 = 0;
    var n_out: usize = 0;
    while (!engine.isFinished()) {
        _ = try engine.step();
        for (engine.lastEmitted()) |e| {
            if (n_out == 0) t_first = std.time.nanoTimestamp();
            n_out += 1;
            const tk = e.token;
            if (tk == engine.eos) continue;
            const s = try tok.decode(&[_]i32{tk});
            defer alloc.free(s);
            try w.print("{s}", .{s});
            try w.flush();
        }
    }
    const t_end = std.time.nanoTimestamp();
    const ttft_ms = @as(f64, @floatFromInt(t_first - t0)) / 1e6;
    const dec_ms = @as(f64, @floatFromInt(t_end - t_first)) / 1e6;
    try w.print("\n---\n生成 {d} token | TTFT {d:.2} ms | decode {d:.2} ms/token ({d:.1} tok/s)\n", .{
        n_out, ttft_ms, if (n_out > 1) dec_ms / @as(f64, @floatFromInt(n_out - 1)) else 0, if (n_out > 1) @as(f64, @floatFromInt(n_out - 1)) * 1e3 / dec_ms else 0,
    });
    try w.print("token ids:", .{});
    for (seq.token_ids.items[seq.num_prompt_tokens..]) |t| try w.print(" {d}", .{t});
    try w.print("\n", .{});
    try w.flush();
}

pub const Stats = struct {
    mean: f64 = 0,
    p50: f64 = 0,
    p90: f64 = 0,
    p99: f64 = 0,
    max: f64 = 0,
    n: usize = 0,
};

/// 分位数: 线性插值(同 numpy.percentile 默认).
fn percentile(sorted: []const f64, q: f64) f64 {
    if (sorted.len == 0) return 0;
    if (sorted.len == 1) return sorted[0];
    const pos = q * @as(f64, @floatFromInt(sorted.len - 1));
    const lo: usize = @intFromFloat(@floor(pos));
    const hi = @min(lo + 1, sorted.len - 1);
    const frac = pos - @as(f64, @floatFromInt(lo));
    return sorted[lo] + (sorted[hi] - sorted[lo]) * frac;
}

/// 会就地排序 xs.
pub fn stats(xs: []f64) Stats {
    if (xs.len == 0) return .{};
    std.sort.pdq(f64, xs, {}, std.sort.asc(f64));
    var sum: f64 = 0;
    for (xs) |x| sum += x;
    return .{
        .mean = sum / @as(f64, @floatFromInt(xs.len)),
        .p50 = percentile(xs, 0.5),
        .p90 = percentile(xs, 0.9),
        .p99 = percentile(xs, 0.99),
        .max = xs[xs.len - 1],
        .n = xs.len,
    };
}

/// 一次压测运行的逐请求 token 时间戳收集器.
pub const Recorder = struct {
    alloc: Alloc,
    t0_ns: i128 = 0,
    first: []i128, // 每个请求首 token 时刻(0 = 尚未产出)
    prev: []i128, // 上一个 token 时刻
    count: []usize,
    itl_ms: std.ArrayListUnmanaged(f64) = .empty,

    pub fn init(alloc: Alloc, n: usize) !Recorder {
        const r = Recorder{
            .alloc = alloc,
            .first = try alloc.alloc(i128, n),
            .prev = try alloc.alloc(i128, n),
            .count = try alloc.alloc(usize, n),
        };
        @memset(r.first, 0);
        @memset(r.prev, 0);
        @memset(r.count, 0);
        return r;
    }

    pub fn deinit(self: *Recorder) void {
        self.alloc.free(self.first);
        self.alloc.free(self.prev);
        self.alloc.free(self.count);
        self.itl_ms.deinit(self.alloc);
    }

    pub fn start(self: *Recorder, now: i128) void {
        self.t0_ns = now;
    }

    pub fn token(self: *Recorder, req: usize, now: i128) !void {
        if (self.count[req] == 0) {
            self.first[req] = now;
        } else {
            try self.itl_ms.append(self.alloc, @as(f64, @floatFromInt(now - self.prev[req])) / 1e6);
        }
        self.prev[req] = now;
        self.count[req] += 1;
    }

    /// TTFT 样本(ms), 调用方负责 free.
    pub fn ttftMs(self: *const Recorder, alloc: Alloc) ![]f64 {
        const out = try alloc.alloc(f64, self.first.len);
        for (self.first, 0..) |f, i| out[i] = @as(f64, @floatFromInt(f - self.t0_ns)) / 1e6;
        return out;
    }
};

    try w.print("{s}\n", .{js});
    try w.flush();
    if (o.out) |p| try std.fs.cwd().writeFile(.{ .sub_path = p, .data = js });
}

对 llmgen 的预期

RSS：起点把 1.5 GB 的文件整个读进堆，RSS 至少 1.5 GB。改完后进程堆里只有头部（16 MB）、32 MB 暂存区和各种小结构，预期降到 500–600 MB 推算（其中大部分是 CUDA 运行时自己占的）。

加载：目标 0.77 s，其中约 0.35 s 是录 CUDA Graph。读盘部分取决于文件是否在 page cache 里：热的时候 1.5 GB 约 0.3 s。第一次冷读会慢得多，压测前先 cat model.gguf > /dev/null 一次。

吞吐：换掉 DebugAllocator 对 S1 可能有 1–3% 的影响 推算，对大批场景没有。

常见错误

现象原因和处理
error.OutOfBounds 在解析头部时读进来的前缀不够长。lab 的写法是遇到这个错误就把长度翻倍重读；如果你的 parse 用的是别的错误名，对应改 if (e != …) 那一行。
tokenizer 初始化崩溃tokenizer 引用了 parser 缓冲区里的字符串。头部缓冲区（buffer）要活到 tokenizer 析构之后，不能解析完就释放。
error.ShortReadfile_off 算错：数据区偏移是 data_offset + info.offset，data_offset 是对齐之后的值。
bench 的吞吐比第 19 步低检查是不是每个 token 都在 flush 输出，或者 Recorder 的 token() 被放进了内层循环之外的地方重复调用。
peak_rss_mb 是 0macOS 没有 /proc，lab 的 peakRssMb 有 getrusage 的回退分支，一起搬。

阶段 5 · 连续批处理

批量前向 execute(entries)

变长打包，一步里跑多条序列。

本步目标多条序列在同一步里一起前向。forward 从第 04 步起就按 Entry 列表写的，kernel 从第 06 步起就是变长批的，这一步只是把上层接起来：Engine 负责排队和回填，Executor.execute 负责把超大的一步切成多趟。

原理

为什么批处理能提高吞吐

decode 是带宽受限的：一步的时间主要花在把 1.2 GB 权重读一遍。批大小从 1 变成 32，权重还是只读一遍：GEMM 的 M 从 1 变成 32，每读到一个权重，和 32 行输入各乘一次。计算量是 32 倍，但 GPU 的算力在 M = 1 时几乎是闲着的。

所以一步的时间只涨一点，产出的 token 数是 32 倍：

批到 256 时一步要 27 ms，这时候瓶颈换了：每条序列有自己的 KV 要读，256 条序列 × 平均 1000 token 上下文 × 112 KiB ≈ 29 GB 的 KV 流量，大头成了注意力。nsys 里 S3 的 kernel 时间 68% 是 attn_decode_kernel nsys 实测。

变长打包：不要 padding

PyTorch 风格的批是 [B, max_len, hidden]，短序列补 padding。这里不这么做。一步里所有序列的 token 首尾相接排成一行：

序列 A（prefill 5）
a0a1a2a3a4
序列 B（decode）
b7
序列 C（prefill 3）
c0c1c2
这一步的 T = 9
a0a1a2a3a4b7c0c1c2
cu_q
0569
张量形状是 [1, T, …]。哪几行属于哪条序列，记在元数据的 cu_q 里。

大多数 kernel 根本不关心序列边界：Linear、RMSNorm、add、mul、silu、RoPE、embedding 都是逐行算的，T 行来自一条序列还是 256 条，对它们没有区别。只有三处需要知道边界，各自从元数据里读：

kernel需要知道什么读哪个字段
store_kv每个 token 写到哪slots[T]
注意力每条序列的块表、上下文长度、query 范围block_tables[S]、ctx[S]、cu_q[S+1]、tile 表
gather_last每条序列最后一行在哪last[S]
sample每条序列的温度temps[S]

这就是第 04 步那个元数据包、第 05 步 token-major、第 08 步 DYN_S 的全部意义。现在 forward 的代码一行都不用改，传进来的 Entry 多于一条就是批处理。

一步太大：切成多趟

激活 token 区按 max_tokens（8192）分配。但调度器一步可以排进 16384 个 prefill token（max_num_batched_tokens）。超过时 execute 把 Entry 列表切成多趟，每趟不超过上限：

executor.zig · execute / flushPass lab 参考（3090 实测过）
    /// 跑一步. 返回 entries 里每一条的采样 token(按 entries 顺序; 不 emits 的条目其值无意义).
    /// token 数超过单趟上限时切成多趟(条目的 query 段可被切开, 只有末段的采样结果有意义).
    pub fn execute(self: *Executor, entries: []const sch.Entry) ![]const i32 {
        var total: usize = 0;
        for (entries) |e| total += e.q_len;
        if (total <= self.lim.max_tok) return self.forward(entries);

        try self.pass_out.resize(self.alloc, entries.len);
        self.pass_group.clearRetainingCapacity();
        self.pass_owner.clearRetainingCapacity();
        var room = self.lim.max_tok;
        for (entries, 0..) |e, idx| {
            var qs = e.q_start;
            var left = e.q_len;
            while (left > 0) {
                const take = @min(left, room);
                try self.pass_group.append(self.alloc, .{ .seq = e.seq, .q_start = qs, .q_len = take });
                try self.pass_owner.append(self.alloc, idx);
                qs += take;
                left -= take;
                room -= take;
                if (room == 0) {
                    try self.flushPass(entries);
                    room = self.lim.max_tok;
                }
            }
        }
        try self.flushPass(entries);
        return self.pass_out.items;
    }

    fn flushPass(self: *Executor, entries: []const sch.Entry) !void {
        if (self.pass_group.items.len == 0) return;
        const toks = try self.forward(self.pass_group.items);
        for (self.pass_group.items, self.pass_owner.items, toks) |piece, owner, tk| {
            const e = entries[owner];
            if (piece.q_start + piece.q_len == e.q_start + e.q_len) self.pass_out.items[owner] = tk;
        }
        self.pass_group.clearRetainingCapacity();
        self.pass_owner.clearRetainingCapacity();
    }

一条 Entry 可以被切开：前半段在第一趟，后半段在第二趟。这之所以正确，是因为 prefill 注意力读的是 KV cache（第 14 步）：第二趟的 query 能看到第一趟已经写进去的 K/V。只有末段的采样结果有意义（flushPass 里那个判断）。

Engine：一步的三个动作

engine.zig lab 参考（3090 实测过）
pub const Engine = struct {
    alloc: Alloc,
    executor: *exe.Executor,
    cfg: EngineConfig,
    bm: bmm.BlockManager,
    sched: sch.BatchScheduler,
    seqs: std.ArrayListUnmanaged(*sq.Sequence) = .empty,
    next_id: u64 = 0,
    last: []const sch.Entry = &.{},
    emitted: std.ArrayListUnmanaged(Emitted) = .empty,
    eos: i32,

    /// 本步产出: 序列 + 新 token(postprocess 之后 Entry.emits() 已不再成立, 故在此单独记录).
    pub const Emitted = struct { seq: *sq.Sequence, token: i32 };

    /// 堆分配: BatchScheduler 持有指向 self.bm 的指针, Engine 地址必须稳定.
    pub fn init(alloc: Alloc, executor: *exe.Executor, cfg: EngineConfig) !*Engine {
        const self = try alloc.create(Engine);
        errdefer alloc.destroy(self);
        self.* = .{
            .alloc = alloc,
            .executor = executor,
            .cfg = cfg,
            .bm = try bmm.BlockManager.init(alloc, cfg.num_kv_blocks, cfg.block_size),
            .sched = undefined,
            .eos = cfg.eos,
        };
        self.sched = sch.BatchScheduler.init(alloc, .{
            .max_num_seqs = cfg.max_num_seqs,
            .max_num_batched_tokens = cfg.max_num_batched_tokens,
            .policy = cfg.policy,
            .chunk_tokens = cfg.chunk_tokens,
            .eos = cfg.eos,
        }, &self.bm);
        return self;
    }

    pub fn deinit(self: *Engine) void {
        for (self.seqs.items) |s| {
            s.deinit(self.alloc);
            self.alloc.destroy(s);
        }
        self.seqs.deinit(self.alloc);
        self.emitted.deinit(self.alloc);
        self.sched.deinit();
        self.bm.deinit();
        self.alloc.destroy(self);
    }

    pub fn addRequest(self: *Engine, prompt: []const i32, params: sq.SamplingParams) !*sq.Sequence {
        if (prompt.len + params.max_tokens > self.cfg.max_model_len) return error.SequenceTooLong;
        const s = try self.alloc.create(sq.Sequence);
        errdefer self.alloc.destroy(s);
        s.* = try sq.Sequence.init(self.alloc, self.next_id, prompt, params);
        self.next_id += 1;
        try self.seqs.append(self.alloc, s);
        try self.sched.add(s);
        return s;
    }

    pub fn isFinished(self: *const Engine) bool {
        return self.sched.isFinished();
    }

    /// 跑一步; 返回本步结束的序列数. 本步条目见 lastEntries().
    pub fn step(self: *Engine) !usize {
        const entries = try self.sched.schedule();
        self.last = entries;
        if (entries.len == 0) return error.SchedulerStuck;
        const toks = try self.executor.execute(entries);
        // executor 按 entries 顺序给出每条的采样值; postprocess 只要 emits() 的那些(按顺序)
        self.emitted.clearRetainingCapacity();
        var buf: std.ArrayListUnmanaged(i32) = .empty;
        defer buf.deinit(self.alloc);
        for (entries, 0..) |e, i| {
            if (e.emits()) {
                try buf.append(self.alloc, toks[i]);
                try self.emitted.append(self.alloc, .{ .seq = e.seq, .token = toks[i] });
            }
        }
        return self.sched.postprocess(entries, buf.items);
    }

schedule() 排出这一步的 Entry 列表 → execute() 前向并采样 → postprocess() 把新 token 追加到各序列、判断结束、归还块。调用方在每步之后看 lastEmitted() 就知道哪条序列刚出了哪个 token。

/// 一轮请求同时提交, 跑到全部结束。返回墙钟纳秒。
fn runRound(engine: *Engine, reqs: []const workload.Request) !u64 {
    var timer = try std.time.Timer.start();
    for (reqs) |r| {
        _ = try engine.addRequest(r.prompt_ids, .{
            .temperature = r.temperature,
            .max_tokens = r.max_tokens,
            .ignore_eos = r.ignore_eos,
        });
    }
    while (!engine.isFinished()) _ = try engine.runStep();
    return timer.read();
}

对 llmgen 的预期

这是整份教程里吞吐提升最大的一步：S3 是 S1 的 13.9 倍。前面 17 步把单步做到了 2.4 ms，这一步让每一步多干活。

对 llmgen：kernel 和执行层都和 lab 相同了，预期四个场景的吞吐都落在目标的 ±10% 以内 推算。差得多的话，按这个顺序查：

  1. S1 对不对（不对就是前面的问题，不是批处理的问题）；
  2. S4 对不对（批 4–8，走 cuBLAS 路和 CUDA Graph 桶 4 / 8）；
  3. S3 对不对（大批，看有没有抢占、有没有切多趟）。

单趟上限对吞吐不敏感：2048 和 16384 只差 1%。它影响的是激活区占多少显存（44 MB 对 352 MB），省下来的给 KV。

现象原因和处理
error.SchedulerStuckschedule() 返回了空列表但还有序列没结束。多半是一条序列需要的块比总块数还多，或者 max_num_batched_tokens 小于一条 prompt 的长度（prefill_first 要求一条 prompt 能一步放下）。
8 份相同 prompt 的输出互不相同序列之间串了。先查 last[s]（gather 取错行），再查块表（bt + s * max_blocks 的行偏移）。

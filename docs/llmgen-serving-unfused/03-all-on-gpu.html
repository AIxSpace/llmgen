阶段 1 · 去掉主机开销

全图上 GPU

embedding 搬上 GPU，tied 权重只存一份。

本步目标把 embedding 搬到 GPU 上做，input_ids 直接放进显存。做完之后图里只剩两条拷贝边：RoPE 表（只算一次）和 logits 回拷（第 09 步去掉）；CPU 上那份 311 MB 的词表权重也不用再存。

原理

现在 embedding 是怎么跑的

调度器把所有 layer_id < 0 的全局节点放在 CPU：

src/graph/scheduler.zig 起点（你现在的代码）
    pub fn assignGlobalNodes(self: *Self, device_type: DeviceType) void {
        for (self.compute_graph.tensors.items) |t| {
            if (t.layer_id < 0) {
                t.device_type = device_type;
            }
        }
    }

而 ops.zig 里 embedding 的派发写死了 CPU 特化：

src/ops.zig 起点（你现在的代码）
    pub fn embedding(out_tensor: *Tensor, device_id: i32) !void {
        try EmbeddingImpl(.CPU).execute(device_id, out_tensor);
    }

于是每一步是这样的：

embeddingCPU：逐元素 elemF32 / storeF32
→
copy_embd拷贝代理：H2D
→
28 层GPU
→
logitsGPU
→
copy_logits拷贝代理：D2H
→
sampleCPU

代价有三样：

  • 一份多余的权重。embedding 和 lm_head 用的是同一张 token_embd.weight（tied）。embedding 在 CPU、lm_head 在 GPU，同一张表存了两份：起点日志里 CPU 的 weight 池是 297 MiB llmgen 起点实测。
  • 每步一次 H2D。CPU 算完的 embedding 要经拷贝代理搬上显存。
  • CPU 上的逐元素循环。elemF32 / storeF32 每个元素做一次 dtype 分支，prefill 512 token 就是 52 万次。
src/backend/cpu/embedding.zig 起点（你现在的代码）
const std = @import("std");
const common = @import("common.zig");

const types = @import("../../types.zig");
const Tensor = @import("../../Tensor.zig");

pub const EmbeddingImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const token_ids = out_tensor.src[0].?;
        const weight = out_tensor.src[1].?;

        // [vocab, hidden]
        const vocab_size = common.dimAt(weight, 0);
        const hidden = common.dimAt(weight, 1);

        const seq_len = common.dimAt(out_tensor, 1);

        const token_ids_array = common.i32DataPtr(token_ids);

        for (0..seq_len) |s| {
            const id = token_ids_array[s];
            if (id < 0 or id >= @as(i32, @intCast(vocab_size))) return error.TokenOutOfRange;
            const row: usize = @as(usize, @intCast(id)) * hidden;
            for (0..hidden) |h| {
                common.storeF32(out_tensor, s * hidden + h, common.elemF32(weight, row + h));
            }
        }
    }
};

embedding 在 GPU 上是什么

embedding 查表就是按行拷贝：out[t] = W[ids[t]]。一行 1024 个 bf16 = 2 KB。kernel 里一个 block 负责一个 token，128 个线程每人搬一个 16 字节的 bf16x8，正好一行。

唯一的要求是 ids 也得在显存里。这一步先给它单独开一块常驻缓冲，每步 H2D 几个字节；第 04 步会把它并进统一的元数据包。

typedef __nv_bfloat16 bf16;

__device__ __forceinline__ float bf2f(bf16 x) { return __bfloat162float(x); }
__device__ __forceinline__ bf16 f2bf(float x) { return __float2bfloat16(x); }
// 把 float 舍入到 bf16 精度再转回(复刻 PyTorch 在 bf16 张量之间的中间舍入)

// 8 个 bf16 = 16 字节的向量化读写
struct __align__(16) bf16x8 { __nv_bfloat162 h[4]; };

__device__ __forceinline__ void unpack8(const bf16x8 &v, float *f) {
#pragma unroll
    for (int i = 0; i < 4; ++i) {
        float2 t = __bfloat1622float2(v.h[i]);
        f[2 * i] = t.x;
        f[2 * i + 1] = t.y;
    }
}
__device__ __forceinline__ bf16x8 pack8(const float *f) {
    bf16x8 v;
#pragma unroll
    for (int i = 0; i < 4; ++i) v.h[i] = __floats2bfloat162_rn(f[2 * i], f[2 * i + 1]);
    return v;
}

__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
    return v;
}
__device__ __forceinline__ float warp_max(float v) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) v = fmaxf(v, __shfl_xor_sync(0xffffffffu, v, o));
    return v;
}

// 块内求和(blockDim.x 为 32 的倍数, ≤ 1024); 结果广播给所有线程
__device__ __forceinline__ float block_sum(float v) {
    __shared__ float sh[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    v = warp_sum(v);
    __syncthreads();
    if (lane == 0) sh[wid] = v;
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    float r = (lane < nw) ? sh[lane] : 0.f;
    return warp_sum(r);
}

// lab: module-serving-unfused -- embedding.cu -- token embedding 查表(tied 权重, BF16)
// 对照 module-style backend/cpu/embedding.zig(EmbeddingImpl); C++ 的 CUDA 后端没有单独的 embedding.cu, 这里单列便于对照.
#include "cuda_common.cuh"

// ------------------------------------------------------------------
// embedding: out[t] = W[ids[t]]
// ------------------------------------------------------------------
__global__ void embedding_kernel(bf16x8 *out, const int *ids, const bf16x8 *w, int H8) {
    int t = blockIdx.x;
    long id = ids[t];
    for (int i = threadIdx.x; i < H8; i += blockDim.x) out[(long)t * H8 + i] = w[id * H8 + i];
}

extern "C" void cuda_embedding(cuda_bf16_t *out, const int *ids, const cuda_bf16_t *w, int T, int H) {
    if (T <= 0) return;
    int H8 = H / 8;
    embedding_kernel<<<T, 128, 0, g_cuda_stream>>>((bf16x8 *)out, ids, (const bf16x8 *)w, H8);
}

const Tensor = @import("../../Tensor.zig");
const cu = @import("cuda.zig");

/// src[0] = input_ids [1, T] i32(显存), src[1] = W [vocab, hidden] bf16(显存) → out [1, T, hidden] bf16
pub const EmbeddingImpl = struct {
    pub fn execute(device_id: i32, out_tensor: *Tensor) !void {
        _ = device_id;
        const ids = out_tensor.src[0].?;
        const w = out_tensor.src[1].?;
        if (w.data_type != .GGML_TYPE_BF16 or out_tensor.data_type != .GGML_TYPE_BF16) return error.EmbeddingNeedsBf16;
        cu.cuda_embedding(out_tensor.data, ids.data, w.data, @intCast(out_tensor.dims[1]), @intCast(w.dims[1]));
    }
};

对 llmgen 的预期

这一步的收益主要在内存和结构上，速度上是小头：

  • RSS 降约 300 MB：CPU 不再存词表。
  • decode 每步：少一次 CPU 查表和一次 2 KB 的 H2D，换成 4 字节 H2D 加一个 kernel。预期 ITL 变化在 0.1 ms 以内 推算。
  • prefill：512 token 时 CPU 上 52 万次逐元素读写没了，预期 TTFT 降几毫秒 推算。

更重要的是它让“整张图在一台设备上”成立，后面的 CUDA Graph（第 17 步）要求一步之内不能有 host 参与的计算。

CPU 支线

CPU：布局相关的算子

embedding、rope、attention、gather_last、sample。

本步目标把 CPU 上五个和“数据怎么排”有关的算子换成新布局的版本：embedding、rope、attention、gather_last、sample。它们分别对应主线第 03、05、06、08、09 步；主线做到哪一步，就来这里做对应的一节。

原理

这五个算子本身都不复杂，在 CPU 上就是几十行循环。要点只有一个：读写的布局和元数据必须和 CUDA 版完全相同，因为它们共用同一个执行层、同一个 KV cache 布局、同一份元数据包。

算子读什么元数据布局约定
embedding—（ids 是输入张量，在元数据包里）out[t] = W[ids[t]]，行宽 hidden
ropepos[T]x 是 token-major [T, n_head, hd]
attentionslots[T]、block_tables[S]、ctx[S]、cu_q[S+1]KV cache [块][kv_head][slot][hd]
gather_lastlast[S]out[s] = x[last[s]]
sampletemps[S]、seedout[s] 是 i32 token id

CUDA 版在元数据里拿到的是显存指针，当参数传给 kernel。CPU 版拿到的是同一个结构里的 host 指针，c.ptrAs(i32, s.pos) 直接当数组用。

embedding（主线第 03 步）

原来的实现逐元素 elemF32 → storeF32，每个元素两次 dtype 分支和一次 bf16 ↔ f32 往返。查表是按位拷贝，一行一个 @memcpy 就够了。

backend/cpu/embedding.zig lab 参考（3090 实测过）
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

pub const EmbError = error{TokenOutOfRange};

/// src[0]=input_ids[T] int32, src[1]=token_embd[H, V] BF16 → out[T, H] BF16.
pub const EmbeddingImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const w = out.src[1].?;
        const H: usize = @intCast(c.dimAt(w, 0));
        const vocab: usize = @intCast(c.dimAt(w, 1));
        const ids = c.ptrAs(i32, out.src[0].?.data);
        const src = c.bf16Of(w);
        const dst = c.bf16Of(out);
        for (0..@intCast(c.g_step.n_tok)) |t| {
            const id: usize = @intCast(ids[t]);
            if (ids[t] < 0 or id >= vocab) return EmbError.TokenOutOfRange;
            @memcpy(dst[t * H ..][0..H], src[id * H ..][0..H]);
        }
    }
};

rope（主线第 05 步，必须）

主线第 05 步把 q / k 改成 token-major [1, T, n, hd]，位置从元数据读。原来的 CPU 实现按 head-major 寻址、位置是 start_pos + s：

src/backend/cpu/rope.zig · ApplyRopeImpl 起点（你现在的代码）
pub const ApplyRopeImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const x = out_tensor.src[0].?; // [n_heads, seq_len, head_dim], bf16/f16/f32
        const cos = out_tensor.src[1].?; // [max_seq_len, rope_dim] F32
        const sin = out_tensor.src[2].?; // [max_seq_len, rope_dim] F32

        const head_dim: usize = @intFromFloat(out_tensor.op_params[0]); // 完整 head_dim
        const rope_dim: usize = @intFromFloat(out_tensor.op_params[1]); // RoPE 维度
        const rope_half = rope_dim / 2;
        const start_pos: usize = @intFromFloat(out_tensor.op_params[2]);
        const B = c.dimAt(x, 0);
        const n_heads = c.dimAt(x, 1);
        const seq_len = c.dimAt(x, 2);

        const cos_data: [*]const f32 = @ptrCast(@alignCast(cos.data.?));
        const sin_data: [*]const f32 = @ptrCast(@alignCast(sin.data.?));
        const cos_stride = rope_dim; // cos/sin 的表步长
        const head_stride = seq_len * head_dim;
        const batch_stride = n_heads * seq_len * head_dim;
        for (0..B) |b| {
            for (0..n_heads) |h| {
                for (0..seq_len) |s| {
                    const pos = start_pos + s;
                    const cos_row = cos_data + pos * cos_stride;
                    const sin_row = sin_data + pos * cos_stride;
                    const row = b * batch_stride + h * head_stride + s * head_dim;
                    for (0..rope_half) |i| {
                        const x0 = c.elemF32(x, row + i);
                        const x1 = c.elemF32(x, row + i + rope_half);
                        const ci = cos_row[i];
                        const si = sin_row[i];
                        c.storeF32(out_tensor, row + i, x0 * ci - x1 * si);
                        c.storeF32(out_tensor, row + i + rope_half, x0 * si + x1 * ci);
                    }
                    for (rope_dim..head_dim) |i| c.storeF32(out_tensor, row + i, c.elemF32(x, row + i));
                }
            }
        }
    }
};

不改的话它会把“第 t 个 token 的第 h 个头”读成别的位置，输出是乱的（不会崩）。lab 的版本：

backend/cpu/rope.zig lab 参考（3090 实测过）
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

const Ctx = struct { out: [*]c.bf16, x: [*]const c.bf16, cs: [*]const f32, pos: [*]const i32, n_head: usize, hd: usize };

fn tokens(ctx: Ctx, t0: usize, t1: usize) void {
    const hd = ctx.hd;
    const half = hd / 2;
    for (t0..t1) |t| {
        const cs = ctx.cs + @as(usize, @intCast(ctx.pos[t])) * hd;
        for (0..ctx.n_head) |h| {
            const base = (t * ctx.n_head + h) * hd;
            for (0..half) |i| {
                const a = c.bf2f(ctx.x[base + i]);
                const b = c.bf2f(ctx.x[base + i + half]);
                ctx.out[base + i] = c.f2bf(a * cs[i] - b * cs[half + i]);
                ctx.out[base + i + half] = c.f2bf(b * cs[i] + a * cs[half + i]);
            }
        }
    }
}

/// src[0]=x[DYN, n_head, hd]; op_params[0]=base(cos/sin 表已按 base 预算, 这里不再读).
pub const ApplyRopeImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const s = &c.g_step;
        c.parallelFor(@intCast(s.n_tok), 8, Ctx{
            .out = c.bf16Of(out),
            .x = c.bf16Of(out.src[0].?),
            .cs = c.ptrAs(f32, s.cos_sin),
            .pos = c.ptrAs(i32, s.pos),
            .n_head = @intCast(c.dimAt(out, 1)),
            .hd = @intCast(c.dimAt(out, 2)),
        }, tokens);
    }
};

// ---- src/backend/cpu/rope.zig: 只换 ApplyRopeImpl, 同文件里的 RopeCacheImpl 不动 ----
const RopeCtx = struct {
    out: [*]c.bf16,
    x: [*]const c.bf16,
    cos: [*]const f32, // [max_seq, hd], 每行前后两半存同一个角度, 只读前半
    sin: [*]const f32,
    pos: [*]const i32, // 本步元数据:每个 token 的位置
    n_head: usize,
    hd: usize,
};

fn ropeTokens(ctx: RopeCtx, t0: usize, t1: usize) void {
    const hd = ctx.hd;
    const half = hd / 2;
    for (t0..t1) |t| {
        const row = @as(usize, @intCast(ctx.pos[t])) * hd;
        for (0..ctx.n_head) |h| {
            const base = (t * ctx.n_head + h) * hd;
            for (0..half) |i| {
                const a = c.bf2f(ctx.x[base + i]);
                const b = c.bf2f(ctx.x[base + i + half]);
                const cs = ctx.cos[row + i];
                const sn = ctx.sin[row + i];
                ctx.out[base + i] = c.f2bf(a * cs - b * sn);
                ctx.out[base + i + half] = c.f2bf(b * cs + a * sn);
            }
        }
    }
}

/// src[0] = x [1, T, n_head, hd] bf16, src[1] = cos / src[2] = sin [max_seq, hd] f32 → out 同形(非原地)
pub const ApplyRopeImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const x = out_tensor.src[0].?;
        const s = &c.g_step;
        c.parallelFor(@intCast(s.n_tok), 8, RopeCtx{
            .out = c.bf16Of(out_tensor),
            .x = c.bf16Of(x),
            .cos = c.ptrAs(f32, out_tensor.src[1].?.data),
            .sin = c.ptrAs(f32, out_tensor.src[2].?.data),
            .pos = c.ptrAs(i32, s.pos),
            .n_head = c.dimAt(x, 2),
            .hd = c.dimAt(x, 3),
        }, ropeTokens);
    }
};

公式和主线第 05 步的 rope_kernel 逐项相同：out[i] = a·cos − b·sin，out[i + half] = b·cos + a·sin，每个输出在 f32 里算完舍入一次。

新版分两步，和 CUDA 版一样：先 storeKv 把本步 k / v 按 slot 写进 cache，再在 cache 上算注意力。

backend/cpu/attention.zig（测试之前的全部） lab 参考（3090 实测过）
const MAX_HD = 256;

/// store_kv: k/v[T, nkv, hd] → cache 的 slot 位置.
pub fn storeKv(k: [*]const c.bf16, v: [*]const c.bf16, slots: [*]const i32, kc: [*]c.bf16, vc: [*]c.bf16, n_tok: usize, nkv: usize, hd: usize, bs: usize) void {
    for (0..n_tok) |t| {
        if (slots[t] < 0) continue;
        const slot: usize = @intCast(slots[t]);
        const blk = slot / bs;
        const off = slot % bs;
        for (0..nkv) |h| {
            const dst = ((blk * nkv + h) * bs + off) * hd;
            const src = (t * nkv + h) * hd;
            @memcpy(kc[dst..][0..hd], k[src..][0..hd]);
            @memcpy(vc[dst..][0..hd], v[src..][0..hd]);
        }
    }
}

pub const Args = struct {
    out: [*]c.bf16,
    q: [*]const c.bf16,
    kc: [*]const c.bf16,
    vc: [*]const c.bf16,
    block_tables: [*]const i32,
    ctx_lens: [*]const i32,
    cu_q: [*]const i32,
    n_seq: usize,
    nq: usize,
    nkv: usize,
    hd: usize,
    max_blocks: usize,
    bs: usize,
    scale: f32,
};

/// token t 属于哪条序列: cu_q 上二分(cu_q[s] ≤ t < cu_q[s+1]).
fn seqOf(a: Args, t: usize) usize {
    var lo: usize = 0;
    var hi: usize = a.n_seq;
    while (hi - lo > 1) {
        const mid = (lo + hi) / 2;
        if (@as(usize, @intCast(a.cu_q[mid])) <= t) lo = mid else hi = mid;
    }
    return lo;
}

fn heads(a: Args, w0: usize, w1: usize) void {
    const hd = a.hd;
    const group = a.nq / a.nkv;
    var acc: [MAX_HD]f32 = undefined;
    for (w0..w1) |w| {
        const t = w / a.nq;
        const h = w % a.nq;
        const s = seqOf(a, t);
        const q_beg: usize = @intCast(a.cu_q[s]);
        const q_len: usize = @as(usize, @intCast(a.cu_q[s + 1])) - q_beg;
        const ctx: usize = @intCast(a.ctx_lens[s]);
        const o = a.out + (t * a.nq + h) * hd;
        if (ctx + (t - q_beg) + 1 <= q_len) { // ctx = 0 的 padding 行: 没有可看的 key
            @memset(o[0..hd], 0);
            continue;
        }
        const n_keys = ctx - q_len + (t - q_beg) + 1; // 因果: 看得到位置 0..pos
        const kvh = h / group;
        const qr = a.q + (t * a.nq + h) * hd;
        const bt = a.block_tables + s * a.max_blocks;
        @memset(acc[0..hd], 0);
        var m: f32 = -std.math.inf(f32);
        var l: f32 = 0;
        for (0..n_keys) |p| {
            const blk: usize = @intCast(bt[p / a.bs]);
            const row = ((blk * a.nkv + kvh) * a.bs + p % a.bs) * hd;
            const score = c.dotBf16(qr, a.kc + row, hd) * a.scale;
            const m_new = @max(m, score);
            const corr = @exp(m - m_new);
            const pw = @exp(score - m_new);
            l = l * corr + pw;
            const vr = a.vc + row;
            for (0..hd) |d| acc[d] = acc[d] * corr + pw * c.bf2f(vr[d]);
            m = m_new;
        }
        for (0..hd) |d| o[d] = c.f2bf(acc[d] / l);
    }
}

/// 分页因果 GQA(不写 KV). 供 PagedAttentionImpl 与单测用.
pub fn attend(a: Args, n_tok: usize) void {
    std.debug.assert(a.hd <= MAX_HD);
    c.parallelFor(n_tok * a.nq, 4, a, heads);
}

/// src[0]=q[T, nq, hd], src[1]=k[T, nkv, hd], src[2]=v[T, nkv, hd] → out[T, nq, hd].
pub const PagedAttentionImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const s = &c.g_step;
        const kv = try c.kvOf(out.layer_id);
        const n_tok: usize = @intCast(s.n_tok);
        const nkv: usize = @intCast(s.n_kv_head);
        const hd: usize = @intCast(s.head_dim);
        const bs: usize = @intCast(s.block_size);
        const kc = c.ptrAs(c.bf16, kv.k);
        const vc = c.ptrAs(c.bf16, kv.v);
        storeKv(c.bf16Of(out.src[1].?), c.bf16Of(out.src[2].?), c.ptrAs(i32, s.slots), kc, vc, n_tok, nkv, hd, bs);
        attend(.{
            .out = c.bf16Of(out),
            .q = c.bf16Of(out.src[0].?),
            .kc = kc,
            .vc = vc,
            .block_tables = c.ptrAs(i32, s.block_tables),
            .ctx_lens = c.ptrAs(i32, s.ctx),
            .cu_q = c.ptrAs(i32, s.cu_q),
            .n_seq = @intCast(s.n_seq),
            .nq = @intCast(s.n_head),
            .nkv = nkv,
            .hd = hd,
            .max_blocks = @intCast(s.max_blocks),
            .bs = bs,
            .scale = s.attn_scale,
        }, n_tok);
    }
};

几个值得看的地方：

  • 一条路径管三种情况。CUDA 版有 decode（split-K）和 prefill（FA2）两个 kernel。CPU 版只有一个 heads：对每个 (token, q 头)，用 cu_q 二分找到它属于哪条序列，算出它能看到的 key 数 n_keys = ctx − q_len + 序列内序号 + 1，然后顺序扫。decode（q_len = 1）、prefill、chunked prefill 的后半段、混批都是这一个公式。
  • 在线 softmax。和主线第 13 步讲的是同一个算法，只是这里一次只来一个 key：维护运行最大值 m、分母 l、加权和 acc，每来一个分数就把旧的按 exp(m − m_new) 缩一下。不需要先存下所有分数。
  • padding 行。ctx = 0 的行没有可看的 key，输出 0。CPU 不录 CUDA Graph，执行层不会产生 padding 行，但测试里覆盖了，行为和 CUDA 一致。
  • 并行粒度。按 (token, q 头) 切，grain = 4。decode 时 16 个头分给 4 个线程；prefill 时 T × 16 个工作单元铺满所有核。acc 是 heads 栈上的数组，每个线程一份。
  • 没有 GQA 共享。CUDA 版一个 block 里同组的 q 头共用一次 KV 读取；这里每个 q 头各读各的。CPU 上 KV 在缓存里，重复读的代价小得多，不值得为它把代码写复杂。

/// x [1, T, H] → out [1, S, H]:每条序列取本步最后一个 token 的行, 行号在元数据的 last[S]
pub const GatherLastImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const H: usize = @intCast(c.g_step.hidden);
        const last = c.ptrAs(i32, c.g_step.last);
        const src_rows = c.bf16Of(out_tensor.src[0].?);
        const dst = c.bf16Of(out_tensor);
        for (0..@intCast(c.g_step.n_seq)) |s| {
            const r: usize = @intCast(last[s]);
            @memcpy(dst[s * H ..][0..H], src_rows[r * H ..][0..H]);
        }
    }
};

sample（主线第 09 步，必须）

和 sample.cu 是同一个算法、同一个随机数：splitmix64(seed, 行, token id) 给出均匀随机数，Gumbel-max 取 argmax；温度 ≤ 0 时是 greedy；并列取小下标。所以同一个 seed 下，CPU 和 CUDA 抽到的 token 在 logits 相同时也相同。

backend/cpu/sample.zig lab 参考（3090 实测过）
const std = @import("std");
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

pub fn splitmix64(x0: u64) u64 {
    var x = x0 +% 0x9E3779B97F4A7C15;
    x = (x ^ (x >> 30)) *% 0xBF58476D1CE4E5B9;
    x = (x ^ (x >> 27)) *% 0x94D049BB133111EB;
    return x ^ (x >> 31);
}

const Ctx = struct { out: [*]i32, logits: [*]const c.bf16, temps: [*]const f32, seed: u64, v: usize };

fn rows(ctx: Ctx, s0: usize, s1: usize) void {
    for (s0..s1) |s| {
        const row = ctx.logits + s * ctx.v;
        const temp = ctx.temps[s];
        const seed = splitmix64(ctx.seed ^ (@as(u64, s) << 40));
        var best = -std.math.inf(f32);
        var best_i: usize = 0;
        for (0..ctx.v) |i| {
            var val = c.bf2f(row[i]);
            if (temp > 0) {
                const r = splitmix64(seed +% i);
                const u = (@as(f32, @floatFromInt(r >> 40)) + 0.5) * (1.0 / 16777216.0); // (0,1)
                val = val * (1 / temp) - @log(-@log(u));
            }
            if (val > best) { // 严格大于: 并列保留小下标
                best = val;
                best_i = i;
            }
        }
        ctx.out[s] = @intCast(best_i);
    }
}

/// src[0]=logits[S, V] BF16 → out[S] int32.
pub const SampleImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const s = &c.g_step;
        const logits = out.src[0].?;
        c.parallelFor(@intCast(s.n_seq), 1, Ctx{
            .out = c.ptrAs(i32, out.data),
            .logits = c.bf16Of(logits),
            .temps = c.ptrAs(f32, s.temps),
            .seed = c.ptrAs(u64, s.seed)[0],
            .v = @intCast(c.dimAt(logits, 1)),
        }, rows);
    }
};

// ============================================================
// 测试: 与朴素 f32 参考对拍(分页 + GQA + 因果 + 两条序列混批 + padding 行)
// ============================================================
const testing = std.testing;

fn refOne(q: []const c.bf16, kc: []const c.bf16, vc: []const c.bf16, bt: []const i32, nq: usize, nkv: usize, hd: usize, bs: usize, t_q: usize, h: usize, n_keys: usize, out: []f32) void {
    const kvh = h / (nq / nkv);
    const scores = testing.allocator.alloc(f32, n_keys) catch unreachable;
    defer testing.allocator.free(scores);
    var mx: f32 = -std.math.inf(f32);
    for (0..n_keys) |p| {
        const blk: usize = @intCast(bt[p / bs]);
        const row = ((blk * nkv + kvh) * bs + p % bs) * hd;
        var dot: f32 = 0;
        for (0..hd) |d| dot += c.bf2f(q[(t_q * nq + h) * hd + d]) * c.bf2f(kc[row + d]);
        scores[p] = dot / @sqrt(@as(f32, @floatFromInt(hd)));
        mx = @max(mx, scores[p]);
    }
    var sum: f32 = 0;
    for (scores) |*sc| {
        sc.* = @exp(sc.* - mx);
        sum += sc.*;
    }
    @memset(out, 0);
    for (0..n_keys) |p| {
        const blk: usize = @intCast(bt[p / bs]);
        const row = ((blk * nkv + kvh) * bs + p % bs) * hd;
        for (0..hd) |d| out[d] += scores[p] / sum * c.bf2f(vc[row + d]);
    }
}

对 llmgen 的预期

这五个算子不是 CPU 上的时间大头（大头是 Linear，见 C4），这一页的目的是保持能跑、保持和 CUDA 对齐。

  • rope：从逐元素 elemF32 变成 bf16 直读 + 按 token 并行。prefill 时快一个数量级以上 推算；decode 时只有 1 个 token，不开线程。
  • attention：单线程变多线程；写 KV 从“逐 token × 逐头 memcpy 进每层自己的块池”变成按 slot 直接写。长 prompt 的 prefill 是 O(T²)，这是 CPU 上除 Linear 外最重的算子。
  • embedding / gather / sample：都是微秒到亚毫秒级，对总时间没有可见影响。

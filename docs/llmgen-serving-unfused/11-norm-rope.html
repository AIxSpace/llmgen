阶段 3 · kernel

RMSNorm 与 RoPE

按行宽选线程映射，舍入点对齐 PyTorch。

本步目标RMSNorm 按行宽用两种线程映射：hidden 行（1024）一个 block 一行，每头的行（128）一个 warp 一行。数值上对齐 PyTorch 的舍入点。顺带回头看第 05 步换上的 RoPE kernel 为什么那样排线程。

原理

RMSNorm 的结构

一行的计算分两趟：先求平方和（归约），再逐元素乘 rstd × w。归约需要一行里所有线程把各自的部分和汇到一起，这决定了线程怎么排。

Qwen3 里 RMSNorm 有两种行宽：

行宽每 token 几行每层几次
attn_norm / ffn_norm / final_normhidden = 102412
q_normhead_dim = 12816（每个 q 头一行）1
k_normhead_dim = 1288（每个 kv 头一行）1

起点：不管多宽，都是一个 warp 一行

src/backend/cuda/norm_warp.cu 起点（你现在的代码）
template<typename T, typename W>
__global__ void rmsnorm_warp_kernel(
    T* __restrict__ out,
    const T* __restrict__ x ,         // [1,seq,hidden]
    const W* __restrict__ weight,     // [hidden]
    int rows, int hidden_size, float eps
){
    // 每个 warp 处理一行
    int warp_id = threadIdx.x / 32; // 因为一个warp是32个线程，所以每个warp处理一行
    int lane_id = threadIdx.x % 32; // 算出当前线程在warp中的编号
    int row = blockIdx.x * (blockDim.x / 32) + warp_id; // 算出当前线程处理的行号，共rows行
    if (row >= rows) return; // 超出行数，直接返回

    T* out_row = out + (long)row * hidden_size;   // 输出地址
    const T* x_row = x + (long)row * hidden_size; // 输入地址

    // ===== Step 1: 计算 sum(x^2) =====
    float sum = 0.f;
    // 32 个线程交错读取整个 hidden_size 维度
    for (int i = lane_id; i < hidden_size; i += 32) {
        float v = float(x_row[i]);
        sum += v * v;
    }

    // warp reduce
    #pragma unroll
    for (int offset = 16; offset > 0; offset /= 2) {
        sum += __shfl_down_sync(0xffffffff, sum, offset);
    }
    float mean_sq = __shfl_sync(0xffffffff, sum, 0) / hidden_size;
    float inv_rms = rsqrtf(mean_sq + eps);
    // ===== Step 2: normalize + scale =====
    for (int i = lane_id; i < hidden_size; i += 32) {
        float v = float(x_row[i]);
        float o = v * inv_rms * float(weight[i]);
        out_row[i] = T(o);
    }
}

32 个 lane 交错读一行。行宽 1024 时每个 lane 循环 32 次，每次读一个 2 字节的 bf16，两趟共读 64 次、写 32 次。行宽 128 时每个 lane 循环 4 次，倒是合适。

lab：按行宽选映射

backend/cuda/normalization.cu lab 参考（3090 实测过）
// lab: module-serving-unfused -- normalization.cu -- RMSNorm
// 对照 module-style backend/cuda/normalization.cu 与 C++ src/backend/cuda/normalization.cu(RmsNormImpl<CUDA>).
// 行宽 = 权重长度: attn_norm / ffn_norm / final_norm 按 hidden(一个 block 一行), q_norm / k_norm 按 head_dim=128(一个 warp 一行).
// 数值同 Qwen3RMSNorm: (x·rstd).to(bf16) · w → bf16.
#include "cuda_common.cuh"

// ------------------------------------------------------------------
// rmsnorm: x 看成 [rows, D], out = (x·rstd).to(bf16) · w
//   D = 128(每头 QK-norm): 一个 warp 一行, 一个 block 4 行
//   其余(hidden):          一个 block 一行
// ------------------------------------------------------------------
__global__ void rmsnorm_row_kernel(bf16x8 *out, const bf16x8 *x, const float *w, int D8, float eps) {
    const long t = blockIdx.x;
    float ss = 0.f;
    for (int i = threadIdx.x; i < D8; i += blockDim.x) {
        float v[8];
        unpack8(x[t * D8 + i], v);
#pragma unroll
        for (int k = 0; k < 8; ++k) ss += v[k] * v[k];
    }
    const float rstd = rsqrtf(block_sum(ss) / (float)(D8 * 8) + eps);
    for (int i = threadIdx.x; i < D8; i += blockDim.x) {
        float v[8];
        unpack8(x[t * D8 + i], v);
#pragma unroll
        for (int k = 0; k < 8; ++k) v[k] = round_bf(v[k] * rstd) * w[i * 8 + k];
        out[t * D8 + i] = pack8(v);
    }
}
__global__ void rmsnorm_head_kernel(bf16 *out, const bf16 *x, const float *w, long rows, float eps) {
    const int HD = 128;
    const long r = (long)blockIdx.x * (blockDim.x >> 5) + (threadIdx.x >> 5);
    const int lane = threadIdx.x & 31;
    if (r >= rows) return;
    const bf16 *xr = x + r * HD;
    float v[4];
    float ss = 0.f;
#pragma unroll
    for (int j = 0; j < 4; ++j) {
        v[j] = bf2f(xr[lane + 32 * j]);
        ss += v[j] * v[j];
    }
    const float rs = rsqrtf(warp_sum(ss) / (float)HD + eps);
    bf16 *o = out + r * HD;
#pragma unroll
    for (int j = 0; j < 4; ++j) o[lane + 32 * j] = f2bf(round_bf(v[j] * rs) * w[lane + 32 * j]);
}

extern "C" void cuda_rmsnorm(cuda_bf16_t *out, const cuda_bf16_t *x, const float *w, long rows, int D, float eps) {
    if (rows <= 0) return;
    if (D == 128) {
        const int rows_per_block = 4;
        rmsnorm_head_kernel<<<(unsigned)((rows + rows_per_block - 1) / rows_per_block), rows_per_block * 32, 0,
                              g_cuda_stream>>>((bf16 *)out, (const bf16 *)x, w, rows, eps);
        return;
    }
    rmsnorm_row_kernel<<<(unsigned)rows, 128, 0, g_cuda_stream>>>((bf16x8 *)out, (const bf16x8 *)x, w, D / 8, eps);
}
  • hidden 行：一个 block（128 线程）一行，每线程一个 bf16x8。128 × 8 = 1024，每个线程只读一次就覆盖整行。平方和用 block_sum 汇总。
  • 每头的行：一个 warp 一行，每个 lane 管 4 个元素（lane, lane+32, lane+64, lane+96），读一次、存在寄存器里，归约后直接写回，不用读第二遍。一个 block 放 4 个 warp。
backend/cuda/cuda_common.cuh lab 参考（3090 实测过）
__device__ __forceinline__ float warp_sum(float v) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) v += __shfl_xor_sync(0xffffffffu, v, o);
    return v;
}
__device__ __forceinline__ float warp_max(float v) {
#pragma unroll
    for (int o = 16; o > 0; o >>= 1) v = fmaxf(v, __shfl_xor_sync(0xffffffffu, v, o));
    return v;
}

// 块内求和(blockDim.x 为 32 的倍数, ≤ 1024); 结果广播给所有线程
__device__ __forceinline__ float block_sum(float v) {
    __shared__ float sh[32];
    int lane = threadIdx.x & 31, wid = threadIdx.x >> 5;
    v = warp_sum(v);
    __syncthreads();
    if (lane == 0) sh[wid] = v;
    __syncthreads();
    int nw = (blockDim.x + 31) >> 5;
    float r = (lane < nw) ? sh[lane] : 0.f;
    return warp_sum(r);
}

warp_sum 用 __shfl_xor_sync 做蝶形归约：5 步之后 32 个 lane 手里都是总和，不经过 shared memory。block_sum 是两级：warp 内 shuffle，warp 之间经一小块 shared memory。

数值：舍入点

PyTorch 的 Qwen3RMSNorm 是 (x * rstd).to(bf16) * w：先把归一化结果舍入到 bf16，再乘权重。lab 的 kernel 照这个写：round_bf(v * rstd) * w。

起点是 v * inv_rms * w 一次算完再舍入，中间少一次舍入。两种写法的结果在部分元素上差 1 ulp。这一步之后 llmgen 的 norm 和 PyTorch / nano-vllm / lab 逐位一致。

回头看 RoPE 的线程映射

第 05 步换 RoPE kernel 时只说了“token-major”，没讲线程为什么那样排。和 norm 是同一个思路：让线程的编号直接对应数据的结构。

RoPE：起点（一线程一元素，每线程 4 次除法 / 取模） 起点（你现在的代码）
template <typename T>
__global__ void apply_rope_kernel(
    T* __restrict__ out,
    const T* __restrict__ input,
    const float* __restrict__ cos_cache,
    const float* __restrict__ sin_cache,
    int64_t B, int64_t n_heads, int64_t seq_len,
    int64_t head_dim, int64_t rope_half, int64_t rope_dim,
    int64_t head_stride, int64_t batch_stride,
    int64_t cos_stride, int64_t start_pos
) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= B * n_heads * seq_len * head_dim) return;

    int i = idx % head_dim;
    int tmp = idx / head_dim;
    int s = tmp % seq_len;
    tmp /= seq_len;
    int h = tmp % n_heads;
    int b = tmp / n_heads;

    const T* row_in = input + b * batch_stride + h * head_stride + s * head_dim;
    T* row_out = out + b * batch_stride + h * head_stride + s * head_dim;
    int64_t pos = start_pos + s;

    if (i < rope_half) {
        float x0 = float(row_in[i]);
        float x1 = float(row_in[i + rope_half]);
        float c = cos_cache[pos * cos_stride + i];
        float si = sin_cache[pos * cos_stride + i];
        row_out[i] = T(x0 * c - x1 * si);
        row_out[i + rope_half] = T(x0 * si + x1 * c);
    } else if (i >= rope_dim) {
        row_out[i] = row_in[i];
    }
}

起点把所有元素摊平，每个线程用除法和取模反推自己是哪个 batch、哪个头、哪个 token、哪一维，再判断自己在前半还是后半。前半的线程写两个元素，后半的线程什么都不干就退出（一半线程是空转的）。

第 05 步的 kernel：blockIdx.x 就是 token，threadIdx.x >> 5 就是头，lane 管 i ∈ {lane, lane+32} 及配对的 i+64。没有除法，没有空转的线程，128 维正好被 32 个 lane × 2 对 × 2 个元素覆盖。

对 llmgen 的预期

llmgen 起点的 norm 已经是 warp 归约，不是 module-style 那种 35.8 µs 的朴素写法，所以这一步的提升有限：

  • decode：每层 4 次 norm，每次只有 1–16 行。kernel 耗时在 2 µs 上下，改前改后差不多。ITL 不变。
  • prefill / 大批：hidden 行从“每 lane 读 32 次 × 2 字节”变成“每线程读 1 次 × 16 字节”，带宽利用率从 80% 左右升到 90% 左右 推算。

这一步的主要价值是数值对齐：从这里开始，llmgen 的 norm 和 lab 逐位相同，离“直接和 lab 比 token id”又近了一步。

error.RmsNormNeedsBf16AndF32Gainnorm 权重不是 F32。少数 GGUF 把 1 维张量也存成 BF16，那就在 loadWeights 里对 norm 权重做一次 bf16 → f32 转换（lab 的 loadOne 里 .f32 分支）。

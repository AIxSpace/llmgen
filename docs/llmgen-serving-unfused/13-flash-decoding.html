阶段 3 · kernel

decode 注意力：split-K

把上下文切成段，喂满显存带宽。

本步目标读懂第 06 步换上的 decode 注意力 kernel：为什么要把上下文切成段（split-K），GQA 组为什么共享一次 KV 读取。这一步不改代码，用回退开关量出它的贡献。

原理

decode 注意力也是带宽问题

decode 时每条序列只有 1 个 query，要和上下文里所有 token 的 K 做点积、再对 V 加权求和。每层要读的 KV：ctx × 8 头 × 128 × 2 B × (K + V)。上下文 768 时每层 3 MB，28 层 88 MB；上下文 4096 时是 470 MB，已经是权重的四成。上下文越长，注意力在 ITL 里占得越多。

起点：一个 warp 扛一整个头

src/backend/cuda/attention.cu（起点，第 06 步已替换） 起点（你现在的代码）
template <typename T>
__global__ void paged_attention_kernel(
    T* __restrict__ out,
    const T* __restrict__ Q,
    int32_t head_dim, int32_t num_q_heads, int32_t seq_len,
    int32_t num_kv_heads, int32_t total_cached,
    float scale_val, int causal, int num_kv_groups,
    const int32_t* __restrict__ page_k_ids,
    const int32_t* __restrict__ page_v_ids,
    int32_t num_blocks,
    const T* __restrict__ k_pool_base,
    const T* __restrict__ v_pool_base,
    int32_t block_stride_elems,
    int32_t q_start
) {
    constexpr int32_t MAX_HEAD_DIM = 256;
    constexpr int32_t MAX_HEAD_DIM_REGS = MAX_HEAD_DIM / 32;
    int32_t head_dim_regs = head_dim / 32;

    int32_t bh = blockIdx.x;
    if (bh >= num_q_heads) return;
    int32_t head_idx  = bh % num_q_heads;
    int32_t batch_idx = bh / num_q_heads;
    int32_t kv_head_idx = head_idx / num_kv_groups;

    int32_t warp_id = threadIdx.x / 32;
    int32_t lane_id = threadIdx.x % 32;
    int32_t q_pos = warp_id + blockIdx.y * (blockDim.x / 32);
    if (q_pos >= seq_len) return;

    int64_t q_base = (int64_t(batch_idx) * num_q_heads + head_idx) * seq_len * head_dim;

    float q_reg[MAX_HEAD_DIM_REGS];
    for (int i = 0; i < head_dim_regs; ++i) {
        int d = lane_id + i * 32;
        q_reg[i] = float(Q[q_base + q_pos * head_dim + d]);
    }

    float m_i = -INFINITY;
    float l_i = 0.f;
    float o_reg[MAX_HEAD_DIM_REGS] = {0};

    constexpr int32_t BLOCK_SIZE = PAGE_BLOCK_SIZE;

    for (int32_t blk = 0; blk < num_blocks; ++blk) {
        int32_t k_blk_id = page_k_ids[blk];
        int32_t v_blk_id = page_v_ids[blk];
        if (k_blk_id < 0 || v_blk_id < 0) break;

        const T* k_blk_ptr = k_pool_base + int64_t(k_blk_id) * block_stride_elems;
        const T* v_blk_ptr = v_pool_base + int64_t(v_blk_id) * block_stride_elems;

        int32_t blk_start = blk * BLOCK_SIZE;
        int32_t blk_end = min(blk_start + BLOCK_SIZE, total_cached);

        for (int32_t p = 0; p < blk_end - blk_start; ++p) {
            int32_t kv_pos = blk_start + p;
            if (causal && kv_pos > q_start + q_pos) continue;

            int64_t kv_off = (int64_t(p) * num_kv_heads + kv_head_idx) * head_dim;

            float score_partial = 0.f;
            for (int i = 0; i < head_dim_regs; ++i) {
                int d = lane_id + i * 32;
                score_partial += q_reg[i] * float(k_blk_ptr[kv_off + d]);
            }
            #pragma unroll
            for (int offset = 16; offset > 0; offset /= 2)
                score_partial += __shfl_down_sync(0xffffffff, score_partial, offset);
            float score = __shfl_sync(0xffffffff, score_partial, 0) * scale_val;

            float m_prev = m_i;
            m_i = fmaxf(m_i, score);
            float exp_score = __expf(score - m_i);
            float exp_prev  = __expf(m_prev - m_i);
            for (int i = 0; i < head_dim_regs; ++i) {
                int d = lane_id + i * 32;
                o_reg[i] = o_reg[i] * exp_prev + exp_score * float(v_blk_ptr[kv_off + d]);
            }
            l_i = l_i * exp_prev + exp_score;
        }
    }

    float inv_l = 1.f / l_i;
    for (int i = 0; i < head_dim_regs; ++i) {
        int d = lane_id + i * 32;
        out[q_base + q_pos * head_dim + d] = T(o_reg[i] * inv_l);
    }
}

template <typename T, int HEAD_DIM=128>
__global__ void flash_attention_warp_kernel(
    T* __restrict__ out,
    const T* __restrict__ Q,
    int64_t B, int64_t num_q_heads, int64_t seq_len, int64_t num_kv_heads,
    float scale_val, int causal, int num_kv_groups,
    const int32_t* __restrict__ page_k_ids,
    const int32_t* __restrict__ page_v_ids,
    const T* __restrict__ k_pool_base,
    const T* __restrict__ v_pool_base,
    int32_t block_stride_elems,
    int32_t total_cached, int32_t q_start
) {
    const int bh = blockIdx.x;
    if (bh >= B * num_q_heads) return;
    const int head_idx  = bh % num_q_heads;
    const int batch_idx = bh / num_q_heads;
    const int kv_head_idx = head_idx / num_kv_groups;

    const int warp_id = threadIdx.x / 32;
    const int lane_id = threadIdx.x % 32;
    const int q_pos = warp_id + blockIdx.y * static_cast<int>(blockDim.x / 32);
    const bool active = q_pos < seq_len; // C++ 在这里 return; 本 lab 的越界 warp 仍要参与搬 tile 与 __syncthreads

    const int64_t q_base  = (batch_idx * num_q_heads + head_idx) * seq_len * HEAD_DIM;
    const int64_t out_base = q_base;

    constexpr int KV_TILE = 32;
    extern __shared__ char smem_[];
    T* shared_K = reinterpret_cast<T*>(smem_);
    T* shared_V = reinterpret_cast<T*>(smem_ + KV_TILE * HEAD_DIM * sizeof(T));

    constexpr int VEC = HEAD_DIM / 32;
    float q_reg[VEC];
    #pragma unroll
    for (int i = 0; i < VEC; ++i) {
        int d_idx = lane_id + i * 32;
        q_reg[i] = active ? float(Q[q_base + q_pos * HEAD_DIM + d_idx]) : 0.f;
    }
    float m_i = -INFINITY;
    float l_i = 0.f;
    float o_reg[VEC] = {0};

    for (int kv_tile = 0; kv_tile < total_cached; kv_tile += KV_TILE) {
        int tile_size = KV_TILE < static_cast<int>(total_cached - kv_tile) ? KV_TILE : static_cast<int>(total_cached - kv_tile);

        // C++: shared_K[i] = K[kv_base + (kv_tile + k) * HEAD_DIM + d]; 这里按逻辑位置查页表取块内 token-major 行
        for (int i = threadIdx.x; i < tile_size * HEAD_DIM; i += blockDim.x) {
            int k = i / HEAD_DIM;
            int d = i % HEAD_DIM;
            int kv_pos = kv_tile + k;
            int blk = kv_pos / PAGE_BLOCK_SIZE;
            int p = kv_pos % PAGE_BLOCK_SIZE;
            int64_t kv_off = (int64_t(p) * num_kv_heads + kv_head_idx) * HEAD_DIM + d;
            shared_K[i] = k_pool_base[int64_t(page_k_ids[blk]) * block_stride_elems + kv_off];
            shared_V[i] = v_pool_base[int64_t(page_v_ids[blk]) * block_stride_elems + kv_off];
        }
        __syncthreads();

        if (active) {
            for (int k = 0; k < tile_size; ++k) {
                int kv_pos = kv_tile + k;
                if (causal && kv_pos > q_start + q_pos) continue;

                float score_partial = 0.f;
                #pragma unroll
                for (int i = 0; i < VEC; ++i) {
                    int d_idx = lane_id + i * 32;
                    score_partial += q_reg[i] * float(shared_K[k * HEAD_DIM + d_idx]);
                }
                #pragma unroll
                for (int offset = 16; offset > 0; offset /= 2) {
                    score_partial += __shfl_down_sync(0xffffffff, score_partial, offset);
                }
                float score = __shfl_sync(0xffffffff, score_partial, 0);
                score *= scale_val;

                float m_prev = m_i;
                m_i = fmaxf(m_i, score);
                float exp_score = __expf(score - m_i);
                float exp_prev  = __expf(m_prev - m_i);
                #pragma unroll
                for (int i = 0; i < VEC; ++i) {
                    int d_idx = lane_id + i * 32;
                    float v = float(shared_V[k * HEAD_DIM + d_idx]);
                    o_reg[i] = o_reg[i] * exp_prev + exp_score * v;
                }
                l_i = l_i * exp_prev + exp_score;
            }
        }
        __syncthreads();
    }

    if (!active) return;
    float inv_l = 1.f / l_i;
    #pragma unroll
    for (int i = 0; i < VEC; ++i) {
        int d_idx = lane_id + i * 32;
        out[out_base + q_pos * HEAD_DIM + d_idx] = T(o_reg[i] * inv_l);
    }
}

// 页表上传(对照 C++ PagedAttentionImpl 里 h_page_k_ids → cudaMalloc + cudaMemcpy 那段)
static void upload_page_table(const int32_t* h_page_k_ids, const int32_t* h_page_v_ids, int32_t num_blocks,
                              int32_t** d_page_k_ids, int32_t** d_page_v_ids) {
    cudaMalloc(d_page_k_ids, num_blocks * sizeof(int32_t));
    cudaMalloc(d_page_v_ids, num_blocks * sizeof(int32_t));
    cudaMemcpy(*d_page_k_ids, h_page_k_ids, num_blocks * sizeof(int32_t), cudaMemcpyHostToDevice);
    cudaMemcpy(*d_page_v_ids, h_page_v_ids, num_blocks * sizeof(int32_t), cudaMemcpyHostToDevice);
}

// 对照 PagedAttentionImpl<Device::CUDA>::execute(Tensor* out, int32_t dev_id) 的 kernel 发射段
extern "C" void cuda_paged_attention(void *out, const void *Q, int dtype,
                                     int32_t head_dim, int32_t num_q_heads, int32_t seq_len, int32_t num_kv_heads,
                                     int32_t total_cached, float scale, int causal, int32_t num_kv_groups,
                                     const int32_t *h_page_k_ids, const int32_t *h_page_v_ids, int32_t num_blocks,
                                     const void *k_pool_base, const void *v_pool_base, int32_t block_stride_elems,
                                     int32_t q_start, int dev_id) {
    cudaSetDevice(dev_id);
    if (head_dim % 32 != 0) {
        fprintf(stderr, "paged_attention requires head_dim to be a multiple of 32\n");
        return;
    }

    int32_t *d_page_k_ids = nullptr, *d_page_v_ids = nullptr;
    upload_page_table(h_page_k_ids, h_page_v_ids, num_blocks, &d_page_k_ids, &d_page_v_ids);

    constexpr int32_t WARPS_PER_BLOCK = 4;
    constexpr int32_t THREADS = WARPS_PER_BLOCK * 32;
    int32_t blocks_x = num_q_heads;
    int32_t blocks_y = (seq_len + WARPS_PER_BLOCK - 1) / WARPS_PER_BLOCK;
    dim3 grid_dim(blocks_x, blocks_y);

    switch (dtype) {
        case GT_F32:
            paged_attention_kernel<float><<<grid_dim, THREADS>>>(
                static_cast<float*>(out), static_cast<const float*>(Q),
                head_dim, num_q_heads, seq_len, num_kv_heads, total_cached,
                scale, causal, num_kv_groups,
                d_page_k_ids, d_page_v_ids, num_blocks,
                static_cast<const float*>(k_pool_base), static_cast<const float*>(v_pool_base),
                block_stride_elems, q_start);
            break;
        case GT_F16:
            paged_attention_kernel<__half><<<grid_dim, THREADS>>>(
                static_cast<__half*>(out), static_cast<const __half*>(Q),
                head_dim, num_q_heads, seq_len, num_kv_heads, total_cached,
                scale, causal, num_kv_groups,
                d_page_k_ids, d_page_v_ids, num_blocks,
                static_cast<const __half*>(k_pool_base), static_cast<const __half*>(v_pool_base),
                block_stride_elems, q_start);
            break;
        case GT_BF16:
            paged_attention_kernel<__nv_bfloat16><<<grid_dim, THREADS>>>(
                static_cast<__nv_bfloat16*>(out), static_cast<const __nv_bfloat16*>(Q),
                head_dim, num_q_heads, seq_len, num_kv_heads, total_cached,
                scale, causal, num_kv_groups,
                d_page_k_ids, d_page_v_ids, num_blocks,
                static_cast<const __nv_bfloat16*>(k_pool_base), static_cast<const __nv_bfloat16*>(v_pool_base),
                block_stride_elems, q_start);
            break;
        default:
            fprintf(stderr, "paged_attention: unsupported dtype\n");
    }

    cudaFree(d_page_k_ids);
    cudaFree(d_page_v_ids);

    cudaError_t err = cudaGetLastError();
    if (err != cudaSuccess) {
        fprintf(stderr, "paged_attention_kernel launch failed: %s\n", cudaGetErrorString(err));
    }
}

grid = (16 个 q 头, 1)，实际干活的是 16 个 warp。每个 warp 串行扫完整个上下文。

3090 有 82 个 SM，每个 SM 能同时跑很多 block。16 个 warp 只占了其中一小角，显存带宽根本喂不满。这和权重 GEMV 不同：GEMV 的输出行数（N = 1024 起）天然提供了足够多的并行单元；decode 注意力在 batch = 1 时只有 16 个头可以并行。

split-K：把上下文切成段

不切分
一个 block 扫 token 0 … 4095
split-K
0–6364–127128–191…4032–4095reduce：合并 64 段
grid = (kv 头, 序列, 段)。每段一个 block，各算各的局部 softmax，最后一个小 kernel 合并。

能这样切，靠的是 softmax 可以分段合并（online softmax）。每段记下三样东西：段内最大值 m、Σ exp(s - m)、未归一化的加权和 o = Σ exp(s - m)·V。合并时把各段换算到同一个全局最大值 M 下：

合并公式 w_p = exp(m_p - M)，输出 = Σ w_p · o_p / Σ w_p · l_p

backend/cuda/attention.cu · attn_decode_reduce_kernel lab 参考（3090 实测过）
__global__ void attn_decode_reduce_kernel(bf16 *out, const float *part_o, const float *part_ml,
                                          const int *ctx_lens, int nq, int part_size, int max_parts) {
    const int h = blockIdx.x, s = blockIdx.y, d = threadIdx.x;
    const int ctx = ctx_lens[s];
    bf16 *o = out + (long)s * nq * HD + (long)h * HD;
    if (ctx <= 0) {
        o[d] = f2bf(0.f);
        return;
    }
    const int np = (ctx + part_size - 1) / part_size;
    const float *ml = part_ml + ((long)s * nq + h) * max_parts * 2;
    const float *po = part_o + ((long)s * nq + h) * max_parts * HD;
    float M = -INFINITY;
    for (int p = 0; p < np; ++p) M = fmaxf(M, ml[2 * p]);
    float num = 0.f, den = 0.f;
    for (int p = 0; p < np; ++p) {
        float w = __expf(ml[2 * p] - M);
        num += w * po[(long)p * HD + d];
        den += w * ml[2 * p + 1];
    }
    o[d] = f2bf(num / den);
}

GQA：一组 q 头共享一次 KV 读取

Qwen3-0.6B 有 16 个 q 头、8 个 kv 头，每 2 个 q 头共用同一份 K/V（G = 2）。起点按 q 头开 block，同一份 KV 被读了两遍。lab 按 kv 头开 block，一个 block 里同时算组内 G 个 q 头：K 读一次，和 G 个 q 各做点积；V 读一次，按 G 组概率各加权一次。KV 的流量减半。

backend/cuda/attention.cu · attn_decode_kernel lab 参考（3090 实测过）
template <int G>
__global__ void __launch_bounds__(128) attn_decode_kernel(
    float *part_o, float *part_ml, const bf16 *q, const bf16 *kc, const bf16 *vc,
    const int *block_tables, const int *ctx_lens, int nq, int nkv, int max_blocks, int block_size,
    float scale, int part_size, int max_parts) {
    const int kvh = blockIdx.x, s = blockIdx.y, part = blockIdx.z;
    const int ctx = ctx_lens[s];
    const int t0 = part * part_size;
    if (t0 >= ctx) return;
    const int n = min(ctx - t0, part_size);
    const int tid = threadIdx.x, lane = tid & 31, wid = tid >> 5;

    extern __shared__ float smem[];
    float *sc = smem;                  // [G][part_size] 分数 → 概率
    float *red = smem + G * part_size; // [8][G][HD] PV 部分和
    __shared__ float q_s[G][HD];
    __shared__ float wred[4][G];
    __shared__ float stat_m[G], stat_l[G];

    for (int i = tid; i < G * HD; i += 128) {
        int g = i / HD, d = i % HD;
        q_s[g][d] = bf2f(q[(long)s * nq * HD + (long)(kvh * G + g) * HD + d]) * scale;
    }
    __syncthreads();

    // ---- phase 1: QK^T. 4 个 lane 一个 token, 每 lane 负责 4 个 16B 分块(32 维) ----
    const int lg = tid & 3, grp = tid >> 2; // 32 组
    float qr[G][32];
#pragma unroll
    for (int g = 0; g < G; ++g)
#pragma unroll
        for (int j = 0; j < 4; ++j)
#pragma unroll
            for (int k = 0; k < 8; ++k) qr[g][j * 8 + k] = q_s[g][(lg + 4 * j) * 8 + k];

    const int *bt = block_tables + (long)s * max_blocks;
    const long head_stride = (long)block_size * HD;
    // 循环次数必须 warp 内一致(下面有全 mask 的 __shfl_xor_sync), 越界 token 用 valid 屏蔽而不是提前退出
    for (int base = 0; base < n; base += 32) {
        const int i = base + grp;
        const bool valid = i < n;
        float acc[G];
#pragma unroll
        for (int g = 0; g < G; ++g) acc[g] = 0.f;
        if (valid) {
            int t = t0 + i;
            long blk = bt[t / block_size];
            const bf16x8 *kr = (const bf16x8 *)(kc + (blk * nkv + kvh) * head_stride + (long)(t % block_size) * HD);
#pragma unroll
            for (int j = 0; j < 4; ++j) {
                float f[8];
                unpack8(kr[lg + 4 * j], f);
#pragma unroll
                for (int g = 0; g < G; ++g)
#pragma unroll
                    for (int k = 0; k < 8; ++k) acc[g] += qr[g][j * 8 + k] * f[k];
            }
        }
#pragma unroll
        for (int g = 0; g < G; ++g) {
            acc[g] += __shfl_xor_sync(0xffffffffu, acc[g], 1);
            acc[g] += __shfl_xor_sync(0xffffffffu, acc[g], 2);
        }
        if (valid && lg == 0)
#pragma unroll
            for (int g = 0; g < G; ++g) sc[g * part_size + i] = acc[g];
    }
    __syncthreads();

    // ---- phase 2: 每个头的 max / exp / sum ----
#pragma unroll
    for (int g = 0; g < G; ++g) {
        float m = -INFINITY;
        for (int i = tid; i < n; i += 128) m = fmaxf(m, sc[g * part_size + i]);
        m = warp_max(m);
        if (lane == 0) wred[wid][g] = m;
    }
    __syncthreads();
    if (tid < G) stat_m[tid] = fmaxf(fmaxf(wred[0][tid], wred[1][tid]), fmaxf(wred[2][tid], wred[3][tid]));
    __syncthreads();
#pragma unroll
    for (int g = 0; g < G; ++g) {
        float m = stat_m[g], l = 0.f;
        for (int i = tid; i < n; i += 128) {
            float p = __expf(sc[g * part_size + i] - m);
            sc[g * part_size + i] = p;
            l += p;
        }
        l = warp_sum(l);
        if (lane == 0) wred[wid][g] = l;
    }
    __syncthreads();
    if (tid < G) stat_l[tid] = wred[0][tid] + wred[1][tid] + wred[2][tid] + wred[3][tid];

    // ---- phase 3: P·V. 16 线程一行(每线程 8 维), 8 个 token 槽并行 ----
    const int dl = tid & 15, ts = tid >> 4;
    float o[G][8];
#pragma unroll
    for (int g = 0; g < G; ++g)
#pragma unroll
        for (int k = 0; k < 8; ++k) o[g][k] = 0.f;
    for (int i = ts; i < n; i += 8) {
        int t = t0 + i;
        long blk = bt[t / block_size];
        const bf16x8 *vr = (const bf16x8 *)(vc + (blk * nkv + kvh) * head_stride + (long)(t % block_size) * HD);
        float f[8];
        unpack8(vr[dl], f);
#pragma unroll
        for (int g = 0; g < G; ++g) {
            float p = sc[g * part_size + i];
#pragma unroll
            for (int k = 0; k < 8; ++k) o[g][k] += p * f[k];
        }
    }
#pragma unroll
    for (int g = 0; g < G; ++g)
#pragma unroll
        for (int k = 0; k < 8; ++k) red[(ts * G + g) * HD + dl * 8 + k] = o[g][k];
    __syncthreads();
    for (int idx = tid; idx < G * HD; idx += 128) {
        int g = idx / HD, d = idx % HD;
        float acc = 0.f;
#pragma unroll
        for (int r = 0; r < 8; ++r) acc += red[(r * G + g) * HD + d];
        long h = kvh * G + g;
        part_o[(((long)s * nq + h) * max_parts + part) * HD + d] = acc;
    }
    if (tid < G) {
        long h = kvh * G + tid;
        float *ml = part_ml + (((long)s * nq + h) * max_parts + part) * 2;
        ml[0] = stat_m[tid];
        ml[1] = stat_l[tid];
    }
}

kernel 分三个阶段，每个阶段的线程映射都不同：

阶段做什么线程怎么排
1Q·Kᵀ：每个 token 一个分数4 个 lane 一个 token，每 lane 读 4 个 bf16x8（32 维）。一轮处理 32 个 token
2段内 max、exp、sum128 个线程交错扫分数数组，warp 归约
3P·V：加权求和16 个线程一行 V（每线程 8 维），8 个 token 槽并行

段长怎么定

backend/cuda/attention.zig lab 参考（3090 实测过）
/// decode split-K 的分段长度: 批越小分段越细, 保证 grid = nkv × S × parts 足够铺满 82 个 SM.
pub fn partSize(s: usize) usize {
    return if (s >= 64) 512 else if (s >= 16) 256 else if (s >= 4) 128 else 64;
}
pub fn maxParts(s: usize, max_model_len: usize) usize {
    const ps = partSize(s);
    return (max_model_len + ps - 1) / ps;
}

批越小，段越细。目标是让 kv 头数 × 序列数 × 段数 这个 block 总数足够铺满 82 个 SM；批大了序列数本身就提供了并行度，段可以长一些，减少 reduce 的开销。

用回退开关量一遍

SV_ATTN_PART=n 把段长的下限设成 n。设成一个很大的数，每条序列就只有 1 段，等于关掉 split-K（GQA 共享还在）：

终端 命令
LLMGEN_WORKLOAD=…/s1.json make llmgen-cuda                       # 默认: split-K
SV_ATTN_PART=100000 LLMGEN_WORKLOAD=…/s1.json make llmgen-cuda   # 每条序列只有 1 段 = 不切分

S1 的上下文只有 512–768，差距不会像 4096 那么大。想看长上下文的效果，把 max_seq_len 调到 4096，自己造一个 3584 → 512 的 workload（lab 的 S5）。

CPU 同步

这一步 CPU 后端要做的事不用动。CPU 上没有 split-K：decode 和 prefill 走同一条朴素路径，每个 (token, q 头) 一个工作单元，按线程并行。KV 在 CPU 缓存里，不存在“block 太少喂不满带宽”的问题。 去看 →

对 llmgen 的预期

lab 的消融：关掉 split-K，S1 慢到 68%（ITL 2.03 → 3.01 ms），长上下文的 S5 慢到 31%（ITL 2.56 → 8.60 ms）。上下文越长，split-K 越重要。

现象原因和处理

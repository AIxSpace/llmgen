阶段 4 · 执行层

活跃区间激活规划

一次规划，任意 T 复用；地址 = 基址 + 偏移 × T。

本步目标激活内存不再每步、每个节点现分配。启动时按“谁在什么时候还活着”规划一次每个节点的偏移，之后任何规模的一步都套同一份规划。结果是两件事：激活池的峰值降下来；同样规模的一步里，每个节点的地址恒定（第 17 步的前提）。

原理

起点：bump 分配，按层回卷

每个节点执行前从激活池顺序切一块；换层时把游标拨回去，下一层覆盖上一层。池的容量要按“最大的那一层的所有激活之和”来定。

它有两个问题：

  • 一层之内不复用。一层里 18 个 kernel 的输出加起来约 60 KB / token，但其中大部分用完马上就没人读了：attn_norm 的输出在 q / k / v 三个 Linear 算完后就死了，gate 在 silu 之后就死了。
  • 地址取决于执行历史。bump 分配的地址是“前面分了多少”的函数。虽然现在每步的结果碰巧相同，但这不是一个可以依赖的性质。

活跃区间

一个激活从它被算出来的那一步开始活，到它最后一个消费者执行完为止。之后它占的内存就可以给别人。

横轴是执行序。同一时刻活着的激活不能重叠，其余位置可以复用。

规划算法是 first-fit：按执行序扫一遍，先释放已经死掉的区间，再给当前节点在现有区间的空隙里找第一个放得下的位置。

关键：按“每行字节”规划，对任意 T 都成立

图里的激活除了尾部几个，都是“每个 token 一行”：大小 = 行字节 × T。如果直接按字节规划，T 一变就得重算。

lab 的做法是把单位换成“每行字节”：规划出每个节点的行内偏移 off，运行时地址 = base + off × T。

为什么不会重叠 两个同时活着的节点 A、B 在规划里满足 off_A + row_A ≤ off_B。两边同乘 T：off_A·T + row_A·T ≤ off_B·T，即 A 的整块在 B 之前结束。对任何 T 都成立。

gather_last 之后的节点（logits、采样结果）每序列一行，单独一个区，地址 = seq_base + off × S。这就是第 08 步引入 DYN_S 的另一个用处：规划器靠它区分节点属于哪个区。

视图节点（拆头的 reshape）不占空间，和根节点同址；视图的消费者算作根的使用点，会延长根的活跃区间。

pub const Scale = enum { tok, seq };

pub const Slot = struct {
    node: *Tensor,
    scale: Scale,
    row_bytes: usize, // 对齐后的「每行」字节(单位偏移)
    off: usize, // 在对应区里的「每行」偏移
    view_of: ?usize = null, // 视图节点: 根节点在 slots 里的下标(不占空间, 地址 = 根的地址)
};

/// 对齐到 256 字节: off × T 仍是 256 的倍数(cuBLAS / 16B 向量化读写都满意).
const ROW_ALIGN: usize = 256;

pub const MemoryPlan = struct {
    alloc: Alloc,
    slots: []Slot, // 与 exec_order 一一对应
    tok_row_bytes: usize, // token 区峰值(每 token 字节)
    seq_row_bytes: usize, // 序列区峰值(每序列字节)

    pub fn deinit(self: *MemoryPlan) void {
        self.alloc.free(self.slots);
    }

    /// 把计划落到具体的 (T, S): 给每个计算节点的 data 赋地址.
    pub fn bind(self: *const MemoryPlan, tok_base: usize, seq_base: usize, n_tok: usize, n_seq: usize) void {
        for (self.slots) |s| {
            if (s.view_of) |r| {
                s.node.data = self.slots[r].node.data; // 执行序: 根一定在视图之前
                continue;
            }
            const addr = switch (s.scale) {
                .tok => tok_base + s.off * n_tok,
                .seq => seq_base + s.off * n_seq,
            };
            s.node.data = @ptrFromInt(addr);
        }
    }

    pub fn tokArenaBytes(self: *const MemoryPlan, max_tok: usize) usize {
        return self.tok_row_bytes * max_tok;
    }
    pub fn seqArenaBytes(self: *const MemoryPlan, max_seq: usize) usize {
        return self.seq_row_bytes * max_seq;
    }
};

/// 行字节 & 所属区: 动态维 DYN → token 区, DYN_S → 序列区; 其余维相乘 × dtype.
fn rowOf(node: *const Tensor) !struct { scale: Scale, bytes: usize } {
    var scale: ?Scale = null;
    var n: usize = 1;
    for (node.dims) |d| {
        if (d == 0) break;
        if (d == of.DYN) {
            scale = .tok;
        } else if (d == of.DYN_S) {
            scale = .seq;
        } else n *= @intCast(d);
    }
    const sc = scale orelse return error.StaticActivationUnsupported;
    return .{ .scale = sc, .bytes = std.mem.alignForward(usize, n * node.dtype.size(), ROW_ALIGN) };
}

const Live = struct { off: usize, size: usize, until: usize };

/// first-fit: 在当前存活区间(按偏移排序)之间找第一个放得下的空隙.
fn place(live: *std.ArrayListUnmanaged(Live), alloc: Alloc, size: usize, until: usize) !usize {
    std.sort.insertion(Live, live.items, {}, struct {
        fn lt(_: void, a: Live, b: Live) bool {
            return a.off < b.off;
        }
    }.lt);
    var cur: usize = 0;
    for (live.items) |l| {
        if (l.off >= cur + size) break;
        cur = @max(cur, l.off + l.size);
    }
    try live.append(alloc, .{ .off = cur, .size = size, .until = until });
    return cur;
}

pub fn build(alloc: Alloc, graph: *const G.ComputeGraph) !MemoryPlan {
    const order = graph.getExecutionOrder();

    // 每个节点在执行序里的位置 + 最后一次被消费的位置
    var pos: std.AutoHashMapUnmanaged(*Tensor, usize) = .empty;
    defer pos.deinit(alloc);
    for (order, 0..) |n, i| try pos.put(alloc, n, i);
    // 视图 → 根节点下标(视图的视图也归到同一个根); 非视图的根是自己
    const root = try alloc.alloc(usize, order.len);
    defer alloc.free(root);
    for (order, 0..) |n, i| {
        root[i] = i;
        if (n.type == .TENSOR_TYPE_VIEW) {
            const si = pos.get(n.src[0] orelse return error.ViewWithoutSource) orelse return error.ViewOfNonActivation;
            root[i] = root[si];
        }
    }
    var last_use = try alloc.alloc(usize, order.len);
    defer alloc.free(last_use);
    for (order, 0..) |_, i| last_use[i] = i;
    for (order, 0..) |n, i| {
        for (n.src) |ms| {
            const s = ms orelse continue;
            if (pos.get(s)) |si| last_use[root[si]] = @max(last_use[root[si]], i);
        }
    }
    // 图输出活到最后(采样结果要拷回 host)
    for (graph.getExternalOutputs()) |o| {
        if (pos.get(o)) |oi| last_use[root[oi]] = order.len;
    }

    const slots = try alloc.alloc(Slot, order.len);
    errdefer alloc.free(slots);
    var live_tok: std.ArrayListUnmanaged(Live) = .empty;
    defer live_tok.deinit(alloc);
    var live_seq: std.ArrayListUnmanaged(Live) = .empty;
    defer live_seq.deinit(alloc);
    var peak_tok: usize = 0;
    var peak_seq: usize = 0;

    for (order, 0..) |n, i| {
        // 释放: 最后使用点 < i 的区间(使用点 == i 的是本节点的输入, 不能与输出重叠)
        inline for (.{ &live_tok, &live_seq }) |lv| {
            var k: usize = 0;
            while (k < lv.items.len) {
                if (lv.items[k].until < i) _ = lv.swapRemove(k) else k += 1;
            }
        }
        if (root[i] != i) {
            const rs = slots[root[i]];
            slots[i] = .{ .node = n, .scale = rs.scale, .row_bytes = 0, .off = rs.off, .view_of = root[i] };
            continue;
        }
        const r = try rowOf(n);
        const lv = if (r.scale == .tok) &live_tok else &live_seq;
        const off = try place(lv, alloc, r.bytes, last_use[i]);
        if (r.scale == .tok) peak_tok = @max(peak_tok, off + r.bytes) else peak_seq = @max(peak_seq, off + r.bytes);
        slots[i] = .{ .node = n, .scale = r.scale, .row_bytes = r.bytes, .off = off };
    }
    return .{ .alloc = alloc, .slots = slots, .tok_row_bytes = peak_tok, .seq_row_bytes = peak_seq };
}

常见错误

现象原因和处理
error.StaticActivationUnsupported规划器遇到了没有动态维的节点。传给 build 的列表里混进了 CACHE 节点（RoPE 表）。只传每步都算的节点。
error.ViewOfNonActivation某个视图的源不在列表里。视图的源必须是每步都算的激活。
某一层之后数值乱两个活着的激活重叠。最常见的原因：有 kernel 读了不是自己 src 的张量（规划器不知道这个依赖）。阶段 1–3 的 kernel 都只读 src 和元数据，不该出现。
OutOfPoolMemoryworkspaceBytes 少算了某块 scratch，或者 RoPE 表的拷贝代理（copy_bytes）没加进去。

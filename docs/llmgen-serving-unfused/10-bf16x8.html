阶段 3 · kernel

bf16x8 向量化

逐元素 kernel 一个线程搬 16 字节。

本步目标add / mul / silu 三个逐元素 kernel 从“一个线程搬一个 bf16（2 字节）”改成“一个线程搬 8 个 bf16（16 字节）”。

原理

逐元素 kernel 是带宽受限的

add、mul、silu 每个元素只做一两次浮点运算，时间全花在读写显存上。这类 kernel 快不快，看的是每秒能搬多少字节，上限是显存带宽（3090 约 936 GB/s）。

起点的写法是最直接的：

src/backend/cuda/arithmetic.cu（add 部分） 起点（你现在的代码）
template<typename T>
__global__ void add_kernel(const T* __restrict__ input1,const T* __restrict__ input2,T* __restrict__ output, size_t size) {
    size_t glob_id = blockIdx.x * blockDim.x + threadIdx.x;
    if (glob_id < size) {
        output[glob_id] = input1[glob_id] + input2[glob_id];
    }
}

// 对照 AddImpl<Device::CUDA>::execute(Tensor* t, int32_t dev_id)
extern "C" void cuda_add(void *out, const void *in1, const void *in2, int dtype, size_t numel, int dev_id) {
    cudaSetDevice(dev_id);

    constexpr int threads = 256;
    int blocks = (numel + threads - 1) / threads;

    switch (dtype) {
        case GT_F16:
            add_kernel<<<blocks, threads>>>(static_cast<const __half*>(in1), static_cast<const __half*>(in2), static_cast<__half*>(out), numel);
            break;
        case GT_BF16:
            add_kernel<<<blocks, threads>>>(static_cast<const __nv_bfloat16*>(in1), static_cast<const __nv_bfloat16*>(in2), static_cast<__nv_bfloat16*>(out), numel);
            break;
        case GT_F32:
            add_kernel<<<blocks, threads>>>(static_cast<const float*>(in1), static_cast<const float*>(in2), static_cast<float*>(out), numel);
            break;
        default:
            fprintf(stderr, "cuda::add not implemented\n");
    }
}

一个线程读两个 bf16、写一个 bf16，每次访存 2 字节。问题不在“算得慢”，而在同样的数据量要发起 8 倍的访存指令。GPU 的访存指令有固定开销，单次搬得太少，带宽就喂不满。

实测：每线程搬多宽合适

bf16x8

backend/cuda/cuda_common.cuh（第 03 步已经加进 llmgen） lab 参考（3090 实测过）
// 8 个 bf16 = 16 字节的向量化读写
struct __align__(16) bf16x8 { __nv_bfloat162 h[4]; };

__device__ __forceinline__ void unpack8(const bf16x8 &v, float *f) {
#pragma unroll
    for (int i = 0; i < 4; ++i) {
        float2 t = __bfloat1622float2(v.h[i]);
        f[2 * i] = t.x;
        f[2 * i + 1] = t.y;
    }
}
__device__ __forceinline__ bf16x8 pack8(const float *f) {
    bf16x8 v;
#pragma unroll
    for (int i = 0; i < 4; ++i) v.h[i] = __floats2bfloat162_rn(f[2 * i], f[2 * i + 1]);
    return v;
}

一个 16 字节对齐的结构体，编译器会生成一条 128 位的 load / store。unpack8 把它解成 8 个 float，在 f32 里算，pack8 就近舍入回 bf16。

前提有两个：元素数是 8 的倍数（hidden 1024、ffn 3072 都是）；地址 16 字节对齐（激活池按 64 对齐分配，满足）。

grid-stride 循环

lab 的 kernel 还有一处不同：线程不是“一人一个元素然后退出”，而是循环处理 i, i + 总线程数, i + 2×总线程数, …。配合 grid_for 给 grid 设上限（65535 × 8 个 block），大批时不会因为 block 数超限而发射失败，同一份 kernel 从 1024 个元素到几亿个元素都能用。

改动

1换掉两个 .cu

用 lab 的 arithmetic.cu 和 activation.cu 覆盖 llmgen 的同名文件：

backend/cuda/arithmetic.cu lab 参考（3090 实测过）
#include "cuda_common.cuh"

// ------------------------------------------------------------------
// add / mul: out = a ∘ b, n 个元素(n 为 8 的倍数)
// ------------------------------------------------------------------
__global__ void add_kernel(bf16x8 *out, const bf16x8 *a, const bf16x8 *b, long n8) {
    for (long i = (long)blockIdx.x * blockDim.x + threadIdx.x; i < n8; i += (long)gridDim.x * blockDim.x) {
        float x[8], y[8];
        unpack8(a[i], x);
        unpack8(b[i], y);
#pragma unroll
        for (int k = 0; k < 8; ++k) x[k] += y[k];
        out[i] = pack8(x);
    }
}
__global__ void mul_kernel(bf16x8 *out, const bf16x8 *a, const bf16x8 *b, long n8) {
    for (long i = (long)blockIdx.x * blockDim.x + threadIdx.x; i < n8; i += (long)gridDim.x * blockDim.x) {
        float x[8], y[8];
        unpack8(a[i], x);
        unpack8(b[i], y);
#pragma unroll
        for (int k = 0; k < 8; ++k) x[k] *= y[k];
        out[i] = pack8(x);
    }
}

extern "C" void cuda_add(cuda_bf16_t *out, const cuda_bf16_t *a, const cuda_bf16_t *b, long n) {
    const long n8 = n / 8;
    if (n8 <= 0) return;
    add_kernel<<<grid_for(n8, 256), 256, 0, g_cuda_stream>>>((bf16x8 *)out, (const bf16x8 *)a, (const bf16x8 *)b, n8);
}
extern "C" void cuda_mul(cuda_bf16_t *out, const cuda_bf16_t *a, const cuda_bf16_t *b, long n) {
    const long n8 = n / 8;
    if (n8 <= 0) return;
    mul_kernel<<<grid_for(n8, 256), 256, 0, g_cuda_stream>>>((bf16x8 *)out, (const bf16x8 *)a, (const bf16x8 *)b, n8);
}
backend/cuda/activation.cu lab 参考（3090 实测过）
#include "cuda_common.cuh"

// ------------------------------------------------------------------
// silu: out = x / (1 + e^-x), n 个元素(n 为 8 的倍数)
// ------------------------------------------------------------------
__global__ void silu_kernel(bf16x8 *out, const bf16x8 *a, long n8) {
    for (long i = (long)blockIdx.x * blockDim.x + threadIdx.x; i < n8; i += (long)gridDim.x * blockDim.x) {
        float x[8];
        unpack8(a[i], x);
#pragma unroll
        for (int k = 0; k < 8; ++k) x[k] = x[k] / (1.f + __expf(-x[k]));
        out[i] = pack8(x);
    }
}

extern "C" void cuda_silu(cuda_bf16_t *out, const cuda_bf16_t *x, long n) {
    const long n8 = n / 8;
    if (n8 <= 0) return;
    silu_kernel<<<grid_for(n8, 256), 256, 0, g_cuda_stream>>>((bf16x8 *)out, (const bf16x8 *)x, n8);
}

对拍验证

add 和 mul 的数值不变：都是在 f32 里算、就近舍入到 bf16，和原来 __nv_bfloat16 的运算符语义相同。silu 把 expf 换成了 __expf（快速版），个别元素可能差 1 个 bf16 ulp。

舍入点为什么重要unfused 的图里 silu 和 mul 是两个节点，silu 的结果先舍入成 bf16 写回显存，mul 再读。如果把两者合成一个 kernel、中间不舍入，26.8% 的元素结果会不同（差 1 ulp）lab 实测。28 层累积之后足以让 argmax 翻转。所以“和参考实现逐 token 一致”要求舍入点一一对应。现在 llmgen 的节点粒度和 lab 相同，舍入点自然对齐。

验收：greedy 输出前 20 个 token 不变，全文通顺。单算子对拍见第 15 步。

对 llmgen 的预期

大数据量下快 1.12 倍。但要说清楚这一步在各场景的实际影响：

  • S1 的 decode：几乎没有影响。一次 add 只有 1024 个元素（2 KB），kernel 本身 1–2 µs，时间花在 launch 上，不在访存上。
  • prefill 和大批：有影响。512 token 的 prefill，一次 silu 是 512 × 3072 × 2 = 3 MB，每层 add ×2、mul ×1、silu ×1 共约 8 MB 读写，28 层 220 MB。S3 一步 8192 token 时是它的 16 倍。

预期 S1 的 ITL 不变，TTFT 降不到 1 ms 推算。这一步主要为第 19 步之后的大批场景铺路。

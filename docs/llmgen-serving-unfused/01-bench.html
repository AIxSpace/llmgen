阶段 0 · 先能量

基准对齐

换成和目标 lab 同一把尺子，再存一份对拍基线。

本步目标用目标 lab 的 S1 口径（512 token 输入、256 token 输出、ignore_eos、5 轮）量出 llmgen 现在的 TTFT、ITL 和吞吐，并存一份 greedy 输出当对拍基线。后面每一步都靠这两样判断“改对了没有”和“快了多少”。

四个场景各量什么

bench/gen_workload.py（S1–S4） lab 参考（3090 实测过）
def s1():
    seed(0)
    rounds = [[{"prompt_ids": [randint(0, 10000) for _ in range(512)], "max_tokens": 256,
                "temperature": 0.6, "ignore_eos": True}] for _ in range(5)]
    return {"name": "s1-single", "rounds": rounds}


def s2():
    seed(1)
    reqs = []
    for _ in range(32):
        prompt = [randint(0, 10000) for _ in range(randint(128, 512))]
        reqs.append({"prompt_ids": prompt, "max_tokens": randint(128, 512), "temperature": 0.6, "ignore_eos": True})
    return {"name": "s2-concurrent32", "rounds": [reqs]}


def s3():
    # 与 nano-vllm bench.py 逐行一致: 先生成全部 prompt, 再生成全部 max_tokens
    seed(0)
    num_seqs, max_input_len, max_ouput_len = 256, 1024, 1024
    prompt_token_ids = [[randint(0, 10000) for _ in range(randint(100, max_input_len))] for _ in range(num_seqs)]
    max_tokens = [randint(100, max_ouput_len) for _ in range(num_seqs)]
    reqs = [{"prompt_ids": p, "max_tokens": m, "temperature": 0.6, "ignore_eos": True}
            for p, m in zip(prompt_token_ids, max_tokens)]
    return {"name": "s3-bench256", "rounds": [reqs]}


def s4():
    # 各轮长度相同(bench 按第 0 轮统计 token 数), 只有 prompt 内容不同
    seed(2)
    lens = [(randint(256, 512), randint(128, 512)) for _ in range(8)]
    rounds = [[{"prompt_ids": [randint(0, 10000) for _ in range(n_in)], "max_tokens": n_out,
                "temperature": 0.6, "ignore_eos": True} for n_in, n_out in lens] for _ in range(3)]
    return {"name": "s4-concurrent8", "rounds": rounds}
场景形状主要看
S11 请求 × 5 轮，512 → 256单流 TTFT、ITL。批处理做完之前只有它能跑
S232 并发，128~512 → 128~512中等并发的吞吐
S3256 并发，100~1024 → 100~1024大批吞吐，时间主要花在 GEMM 和 KV 带宽
S48 并发 × 3 轮decode 批落在 4~8，最考验小批 kernel

每轮 prompt 都不同，temperature=0.6，ignore_eos=True，所以每个引擎的输出 token 数严格相等，吞吐可以直接比。

指标口径

  • TTFT：请求提交到第一个 token 回到 host。包含 prefill 和第一次采样。
  • ITL：同一请求相邻两个 token 的间隔。单请求时 1000 / ITL 就是 decode 吞吐。
  • 输出吞吐：输出 token 总数 ÷ 墙钟时间。墙钟包含 prefill，所以 S1 的吞吐（401.3）略低于 1000 / ITL（408）。

2读 workload 的小模块

src/bench/workload.zig 新建文件
//! bench/workload.zig -- 读 gen_workload.py 生成的 JSON(格式同 module-serving-unfused 的 main.zig)
const std = @import("std");

pub const Request = struct {
    prompt_ids: []const i32,
    max_tokens: usize,
    temperature: f32 = 0.6,
    ignore_eos: bool = true,
};

pub const Workload = struct {
    name: []const u8 = "workload",
    rounds: []const []const Request, // 各轮依次跑, 轮内请求同时提交
};

pub fn load(allocator: std.mem.Allocator, path: []const u8) !std.json.Parsed(Workload) {
    const bytes = try std.fs.cwd().readFileAlloc(allocator, path, 256 << 20);
    defer allocator.free(bytes);
    // .alloc_always: 解析结果不引用 bytes, 上面可以放心 free
    return std.json.parseFromSlice(Workload, allocator, bytes, .{ .ignore_unknown_fields = true, .allocate = .alloc_always });
}

5临时压测入口

src/llmgen.zig 新增
/// 阶段 0 的临时压测入口:按 workload 逐条请求串行跑(批处理要到第 19 步)。
/// 只有 S1 的数字能和目标 lab 直接比;S2–S4 的并发场景这里先不跑。
fn benchSingle(allocator: std.mem.Allocator, executor: *Executor, path: []const u8) !void {
    const workload = @import("bench/workload.zig");
    const pcommon = @import("profiler/common.zig");

    const parsed = try workload.load(allocator, path);
    defer parsed.deinit();
    const wl = parsed.value;

    var ttft_ms: std.ArrayList(f64) = .empty;
    defer ttft_ms.deinit(allocator);
    var itl_ns: std.ArrayList(u64) = .empty;
    defer itl_ns.deinit(allocator);
    var out_tokens: usize = 0;
    var wall_ns: u64 = 0;

    executor.metrics.enabled = false; // 不让 generate 每次打印一屏指标, 下面自己汇总
    for (wl.rounds) |reqs| {
        for (reqs) |r| {
            var timer = try std.time.Timer.start();
            const out = try executor.generate(r.prompt_ids, r.max_tokens, -1, null);
            wall_ns += timer.read();
            defer allocator.free(out);

            out_tokens += out.len;
            try ttft_ms.append(allocator, pcommon.nsToMs(executor.metrics.ttft_ns));
            try itl_ns.appendSlice(allocator, executor.metrics.itl_ns.items);
        }
    }

    var ttft_sum: f64 = 0;
    for (ttft_ms.items) |t| ttft_sum += t;
    var itl_sum: u64 = 0;
    for (itl_ns.items) |t| itl_sum += t;
    const n_itl: f64 = @floatFromInt(@max(itl_ns.items.len, 1));

    std.debug.print(
        "[bench] {s}: 输出 {d} token | 吞吐 {d:.1} tok/s | TTFT mean {d:.2} ms | ITL mean {d:.3} ms p99 {d:.3} ms\n",
        .{
            wl.name,
            out_tokens,
            pcommon.perSec(out_tokens, wall_ns),
            ttft_sum / @as(f64, @floatFromInt(@max(ttft_ms.items.len, 1))),
            pcommon.nsToMs(itl_sum) / n_itl,
            pcommon.nsToMs(pcommon.percentile(itl_ns.items, 99)),
        },
    );
}

6存一份 greedy 基线

性能优化最怕“变快了但算错了”。bf16 下每个张量边界舍入一次，只要计算顺序不变，greedy 输出应该逐 token 相同。

常见错误

现象原因和处理
error.SequenceTooLongworkload 的 512 + 256 超过了 max_seq_len。S1 需要至少 768。
OutOfPoolMemory（激活池）起点的激活池按 max_seq 估算，lm_head 的 [512, 151936] bf16 输出要 148 MB。把 max_seq_len 调到 1024 以上会更大。第 08 步之后这个问题消失。
吞吐比 run.sh 的 15.5 低正常。S1 的吞吐按墙钟算，包含 prefill。
每轮 TTFT 差很多第一轮含 cuBLAS 初始化和 GPU 升频。正式记数前先跑一条短请求预热（第 21 步会加进 bench）。

阶段 5 · 连续批处理

调度策略与抢占

prefill_first 和 chunked：吞吐和延迟的取舍。

本步目标读懂两种调度策略（prefill_first 和 chunked）各自在优化什么、牺牲什么，以及 KV 不够时的抢占。代码第 19 步已经能跑，这一步加一个开关，量出两种策略的差别。

原理

调度器每一步要回答的问题

有一批在等的新请求（waiting），有一批正在生成的序列（running）。这一步让谁进前向、各进多少 token？约束有三个：

  • max_num_seqs：一步最多多少条序列；
  • max_num_batched_tokens：一步最多多少 token；
  • KV 的空闲块够不够。

prefill_first（nano-vllm 的策略）

batch.zig · schedulePrefillFirst lab 参考（3090 实测过）
    fn schedulePrefillFirst(self: *BatchScheduler) !void {
        // ---- prefill ----
        var n_tok: usize = 0;
        while (self.waiting.items.len > 0 and self.scheduled.items.len < self.cfg.max_num_seqs) {
            const seq = self.waiting.items[0];
            const n = seq.len();
            if (n_tok + n > self.cfg.max_num_batched_tokens) break;
            if (!(try self.bm.ensure(seq, n))) break;
            _ = self.waiting.orderedRemove(0);
            seq.status = .running;
            try self.running.append(self.alloc, seq);
            try self.scheduled.append(self.alloc, .{ .seq = seq, .q_start = 0, .q_len = n });
            n_tok += n;
        }
        if (self.scheduled.items.len > 0) return;

        // ---- decode ----
        var i: usize = 0;
        while (i < self.running.items.len and self.scheduled.items.len < self.cfg.max_num_seqs) {
            const seq = self.running.items[i];
            var ok = true;
            while (!self.bm.canEnsure(seq, seq.len())) {
                if (self.running.items.len - 1 > i) {
                    try self.preempt(self.running.pop().?);
                } else {
                    _ = self.running.orderedRemove(i);
                    try self.preempt(seq);
                    ok = false;
                    break;
                }
            }
            if (!ok) break;
            _ = try self.bm.ensure(seq, seq.len());
            try self.scheduled.append(self.alloc, .{ .seq = seq, .q_start = seq.len() - 1, .q_len = 1 });
            i += 1;
        }
    }

有新请求就先 prefill，整条 prompt 一步做完；这一步只做 prefill。没有新请求了才 decode。

  • 好处：简单；prefill 步和 decode 步分开，decode 步全走 CUDA Graph。
  • 坏处：一次大 prefill（最多 16384 token）期间，所有正在 decode 的序列干等一整步。在线服务里表现为 ITL 尖刺。

chunked

batch.zig · scheduleChunked lab 参考（3090 实测过）
    fn scheduleChunked(self: *BatchScheduler) !void {
        var budget = self.cfg.max_num_batched_tokens;
        var prefill_budget = self.cfg.chunk_tokens;
        // ---- running: decode 或未完成 prefill 的续段 ----
        var i: usize = 0;
        while (i < self.running.items.len and self.scheduled.items.len < self.cfg.max_num_seqs and budget > 0) {
            const seq = self.running.items[i];
            const remaining = seq.len() - seq.num_computed_tokens;
            const is_prefill = remaining > 1;
            var q = @min(remaining, budget);
            if (is_prefill) q = @min(q, prefill_budget);
            if (q == 0) {
                i += 1;
                continue;
            }
            const upto = seq.num_computed_tokens + q;
            var ok = true;
            while (!self.bm.canEnsure(seq, upto)) {
                if (self.running.items.len - 1 > i) {
                    try self.preempt(self.running.pop().?);
                } else {
                    _ = self.running.orderedRemove(i);
                    try self.preempt(seq);
                    ok = false;
                    break;
                }
            }
            if (!ok) break;
            _ = try self.bm.ensure(seq, upto);
            try self.scheduled.append(self.alloc, .{ .seq = seq, .q_start = seq.num_computed_tokens, .q_len = q });
            budget -= q;
            if (is_prefill) prefill_budget -= q;
            i += 1;
        }
        // ---- waiting: 新请求的 prompt 分块 ----
        while (self.waiting.items.len > 0 and self.scheduled.items.len < self.cfg.max_num_seqs and budget > 0 and prefill_budget > 0) {
            const seq = self.waiting.items[0];
            const q = @min(seq.len(), @min(budget, prefill_budget));
            if (!(try self.bm.ensure(seq, q))) break;
            _ = self.waiting.orderedRemove(0);
            seq.status = .running;
            seq.num_computed_tokens = 0;
            try self.running.append(self.alloc, seq);
            try self.scheduled.append(self.alloc, .{ .seq = seq, .q_start = 0, .q_len = q });
            budget -= q;
            prefill_budget -= q;
        }
    }

每一步先照顾 running：正在 decode 的各出 1 个 token，prefill 没做完的续一段。剩下的 token 预算再切一块新请求的 prompt 进来，每步新 prefill 的量不超过 chunk_tokens。

  • 好处：decode 每步都能前进，不会被大 prefill 挡住；新请求的第一块能更早开始。
  • 坏处：prefill 和 decode 混在一步里，这一步不是“纯 decode”，走不了 CUDA Graph，注意力全走 prefill 的 kernel；一条长 prompt 要好几步才 prefill 完。

它依赖两件前面已经做好的事：Sequence.num_computed_tokens 记着 prefill 做到哪了；prefill 注意力从 KV cache 读已经算过的部分（第 14 步）。

抢占

batch.zig · preempt lab 参考（3090 实测过）
    fn preempt(self: *BatchScheduler, seq: *Sequence) !void {
        seq.status = .waiting;
        self.bm.deallocate(seq);
        try self.waiting.insert(self.alloc, 0, seq);
        self.num_preemptions += 1;
    }

一条正在 decode 的序列每 16 个 token 要一个新块。空闲块用完时，从 running 队尾（最晚来的）开始牺牲：释放它的全部块，放回 waiting 队首，num_computed_tokens 清零。之后轮到它时从头重新 prefill（包括已经生成的 token）。

抢占很贵：被抢的序列之前的计算全部作废。所以 KV 容量是批处理吞吐的硬约束。

回填

batch.zig · postprocess lab 参考（3090 实测过）
    pub fn postprocess(self: *BatchScheduler, entries: []const Entry, tokens: []const i32) !usize {
        var k: usize = 0;
        var n_finished: usize = 0;
        for (entries) |e| {
            const seq = e.seq;
            const emits = e.emits();
            seq.num_computed_tokens = e.q_start + e.q_len;
            if (!emits) continue;
            const tok = tokens[k];
            k += 1;
            try seq.token_ids.append(self.alloc, tok);
            const hit_eos = !seq.params.ignore_eos and tok == self.cfg.eos;
            if (hit_eos or seq.numCompletionTokens() >= seq.params.max_tokens) {
                seq.status = .finished;
                self.bm.deallocate(seq);
                for (self.running.items, 0..) |r, idx| {
                    if (r == seq) {
                        _ = self.running.orderedRemove(idx);
                        break;
                    }
                }
                n_finished += 1;
            }
        }
        return n_finished;
    }

一条 Entry 只有走到序列末尾（emits()）才产出 token。chunked 策略下 prefill 的中间块不产出。

对比结论
S2，prefill_first → chunked吞吐基本不变（4430.8 → 4407）。平均 TTFT 从 157 ms 降到 101 ms：chunked 下先到的请求不用等所有人一起 prefill 完
S3，prefill_first → chunked吞吐降 4%，ITL p99 从 29 ms 涨到 81 ms，平均 TTFT 从 1.27 s 涨到 1.81 s。离线大批、所有请求同时到达时，chunked 没有优势：混批步走不了 CUDA Graph，prefill 被摊到更多步里
S3，KV 减半抢占 157 次，吞吐降到 4789.8，平均 TTFT 涨到 6.9 秒
KV 减半时块 16 对块 256吞吐相同；块 256 的抢占反而少（139 对 157），但步数更多。差别不大
这些数字说明什么压测用的是“所有请求在 t = 0 同时到达”的离线负载，这对 prefill_first 最有利。chunked 的价值在在线负载：请求陆续到达，新请求的 prefill 不该让老请求的 decode 卡顿。本教程的 bench 测不出这一点，要测得造一个按时间到达的 workload。

对拍验证

两种策略下，同一批 greedy 请求的输出 token 应该相同（数值上只差在 prefill 是一步算完还是分块算，注意力对已缓存部分读的是同样的 K/V）。和第 06 步同样的保留：分块改变了 FA2 里 tile 的切法，累加顺序可能不同，近似并列处允许分叉。

CPU 同步

这一步 CPU 后端要做的事不用动。调度和抢占与设备无关。CPU 的注意力只有一条路径，chunked 的混批步不需要特殊处理。

对 llmgen 的预期

这一步没有“提升”，是一个取舍。按你的场景选：

  • 离线批量、追求总吞吐：prefill_first（默认，也是和 nano-vllm、目标 lab 对比时的口径）。
  • 在线服务、在意已有请求的 ITL 不被新请求打断：chunked，并把 chunk_tokens 调到你能接受的单步时长。

对 llmgen 预期和上表同方向：S2 的平均 TTFT 降三分之一，S3 的吞吐降 4% 左右 推算。

常见错误

现象原因和处理
chunked 下某条序列的输出比 prefill_first 多 / 少 tokenpostprocess 里 num_computed_tokens 要对每条 Entry 都更新（包括不产出 token 的中间块）。
chunked 下卡死（SchedulerStuck）chunk_tokens 设成了 0，或 running 里有序列 remaining = 0。后者说明 num_computed_tokens 多加了。
抢占之后输出错被抢占的序列 num_computed_tokens 必须清零（BlockManager.deallocate 里做了），重新 prefill 时 q_start = 0、长度是 prompt + 已生成。
chunked 下 n_graph_steps 很少正常：只有纯 decode 步才回放图。

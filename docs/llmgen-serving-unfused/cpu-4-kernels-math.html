CPU 支线

CPU：计算类算子

Linear、逐元素、RMSNorm，以及 CPU 算子的单测。

本步目标把 CPU 上四类“纯计算”的算子换掉：Linear、add / mul / silu、RMSNorm。Linear 决定 CPU 的速度，建议最先做；RMSNorm 决定 CPU 和 CUDA 能不能逐 token 相同，必须和主线第 11 步一起做。

Linear（建议最先做；对应主线第 12 步）

原理

CPU 上 Linear 占 99% 的时间（见 C2 的日志）。它和 GPU 上的 GEMV 是同一个问题：decode 时每步要把 1.2 GB 权重读一遍，瓶颈是内存带宽和有多少个核在同时读。

backend/cpu/linear.zig lab 参考（3090 实测过）
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

const Ctx = struct { y: *anyopaque, y_f32: bool, x: [*]const c.bf16, w: [*]const c.bf16, m: usize, n: usize, k: usize };

const M_TILE = 32;

fn cols(ctx: Ctx, n0: usize, n1: usize) void {
    var m0: usize = 0;
    while (m0 < ctx.m) : (m0 += M_TILE) {
        const m1 = @min(m0 + M_TILE, ctx.m);
        for (n0..n1) |j| {
            const wr = ctx.w + j * ctx.k;
            for (m0..m1) |i| {
                const v = c.dotBf16(ctx.x + i * ctx.k, wr, ctx.k);
                if (ctx.y_f32) c.ptrAs(f32, ctx.y)[i * ctx.n + j] = v else c.ptrAs(c.bf16, ctx.y)[i * ctx.n + j] = c.f2bf(v);
            }
        }
    }
}

/// src[0]=x[M, in], src[1]=W[in, out](BF16) → y[M, out](BF16 或 f32).
/// M = 本趟 T(token 区节点)或 S(lm_head 这类序列区节点).
pub const LinearImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const w = out.src[1].?;
        const ctx = Ctx{
            .y = out.data.?,
            .y_f32 = out.dtype == .GGML_TYPE_F32,
            .x = c.bf16Of(out.src[0].?),
            .w = c.bf16Of(w),
            .m = @intCast(c.rowsOf(out)),
            .n = @intCast(c.dimAt(w, 1)),
            .k = @intCast(c.dimAt(w, 0)),
        };
        if (ctx.m == 0) return;
        c.parallelFor(ctx.n, @max(1, 65536 / (ctx.k * ctx.m)), ctx, cols); // 每块至少 ~64K 次乘加
    }
};

三个设计点：

  • 按输出列并行。N 个输出各是一个独立的点积。parallelFor(n, …) 把输出列均分给线程，每个线程读自己那一段权重行，互不干扰。
  • 内层 SIMD。dotBf16 一次处理 VLEN 个 bf16（AVX2 是 8 个）。
  • M 分块。prefill 时 M 可以是几百上千。循环顺序是“块内先 n 再 m”：一行权重（2 KB）读进来后，给块内 32 行输入各用一次，然后才换下一行权重。权重只从内存读一遍；32 行输入共 64 KB，待在 L2 里。如果反过来先 m 再 n，每行输入都要把整个权重矩阵重新读一遍。

grain 的算法 65536 / (k · m) 保证每块至少约 6.5 万次乘加，小矩阵不会被切得太碎。

改动

llmgen 的权重维度是 [N, K]（lab 是 [K, N]），行数从 x.dims[1] 读：

            .y_f32 = out_tensor.data_type == .GGML_TYPE_F32,

            .w = c.bf16Of(w),
            .m = c.dimAt(x, 1),
            .n = c.dimAt(w, 0),
            .k = c.dimAt(w, 1),

数值

f32 累加，最后舍入一次，和 .cu 一样。求和顺序和 CUDA 的 GEMV / cuBLAS 不同，结果有 f32 最后一位的差别，舍入到 bf16 后绝大多数相同。

add / mul / silu（对应主线第 10 步）

原来的实现走通用的 binary(out, fn) + elemF32。换成 bf16 直读直写，大张量时并行：

backend/cpu/arithmetic.zig、activation.zig lab 参考（3090 实测过）
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

const Op = enum { add, mul };
fn Binary(comptime op: Op) type {
    return struct {
        const Ctx = struct { out: [*]c.bf16, a: [*]const c.bf16, b: [*]const c.bf16 };
        fn body(ctx: Ctx, beg: usize, end: usize) void {
            for (beg..end) |i| {
                const x = c.bf2f(ctx.a[i]);
                const y = c.bf2f(ctx.b[i]);
                ctx.out[i] = c.f2bf(switch (op) {
                    .add => x + y,
                    .mul => x * y,
                });
            }
        }
        /// src[0]=a, src[1]=b(同形).
        pub fn execute(out: *Tensor, dev_id: i32) !void {
            _ = dev_id;
            const ctx = Ctx{ .out = c.bf16Of(out), .a = c.bf16Of(out.src[0].?), .b = c.bf16Of(out.src[1].?) };
            c.parallelFor(@intCast(c.numelOf(out)), 32768, ctx, body);
        }
    };
}

/// out = a + b.
pub const AddImpl = Binary(.add);
/// out = a · b.
pub const MulImpl = Binary(.mul);

const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

const Ctx = struct { out: [*]c.bf16, x: [*]const c.bf16 };
fn body(ctx: Ctx, beg: usize, end: usize) void {
    for (beg..end) |i| {
        const x = c.bf2f(ctx.x[i]);
        ctx.out[i] = c.f2bf(x / (1 + @exp(-x)));
    }
}

pub const SiluImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        c.parallelFor(@intCast(c.numelOf(out)), 16384, Ctx{ .out = c.bf16Of(out), .x = c.bf16Of(out.src[0].?) }, body);
    }
};

grain 给得很大（add / mul 32768，silu 16384）：decode 时这些张量只有一两千个元素，一定是串行的；prefill 几百个 token 时才会分给多个线程。

数值和 .cu 逐点相同：add / mul 是精确的 f32 运算加一次舍入；silu 是 x / (1 + exp(−x))，CPU 的 @exp 和 GPU 的 __expf 可能差 1 ulp（f32），舍入到 bf16 后几乎总是相同。

RMSNorm（对应主线第 11 步，必须同步）

src/backend/cpu/norm.zig 起点（你现在的代码）
pub const RmsNormImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const x = out_tensor.src[0].?;
        const gain = out_tensor.src[1].?;

        const r = out_tensor.rank();

        const last = common.dimAt(out_tensor, r - 1);

        const rows = out_tensor.numElements() / last;

        const eps = out_tensor.op_params[0];
        const w_offset = out_tensor.op_params[1];

        for (0..rows) |s| {
            var ss: f32 = 0; // 平方和
            for (0..last) |hidx| {
                const v = common.elemF32(x, s * last + hidx);
                ss += v * v; // 求平方
            }

            const scale = 1.0 / @sqrt(ss / @as(f32, @floatFromInt(last)) + eps);

            for (0..last) |hidx| {
                const v = common.elemF32(x, s * last + hidx) * scale * (common.elemF32(gain, hidx) + w_offset);
                common.storeF32(out_tensor, s * last + hidx, v);
            }
        }
    }
};

原来的写法是 x · scale · gain 一次算完再舍入。主线第 11 步把 CUDA 的舍入点改成了 PyTorch 的 (x · rstd).to(bf16) · w。CPU 不跟着改，两边的 norm 输出就会在部分元素上差 1 ulp，28 层累积后 greedy 输出会分叉。

backend/cpu/normalization.zig lab 参考（3090 实测过）
const T = @import("../../tensor.zig");
const c = @import("common.zig");
const Tensor = T.Tensor;

const Ctx = struct { out: [*]c.bf16, x: [*]const c.bf16, w: [*]const f32, d: usize, eps: f32 };

fn rows(ctx: Ctx, r0: usize, r1: usize) void {
    const d = ctx.d;
    for (r0..r1) |r| {
        const x = ctx.x + r * d;
        var ss: f32 = 0;
        for (0..d) |i| ss += c.bf2f(x[i]) * c.bf2f(x[i]);
        const rstd = 1.0 / @sqrt(ss / @as(f32, @floatFromInt(d)) + ctx.eps);
        const o = ctx.out + r * d;
        for (0..d) |i| o[i] = c.f2bf(c.roundBf(c.bf2f(x[i]) * rstd) * ctx.w[i]);
    }
}

/// src[0]=x(BF16), src[1]=norm 权重(加载期转 f32); op_params[0]=eps.
pub const RmsNormImpl = struct {
    pub fn execute(out: *Tensor, dev_id: i32) !void {
        _ = dev_id;
        const w = out.src[1].?;
        const d: usize = @intCast(c.dimAt(w, 0));
        const n: usize = @intCast(c.numelOf(out));
        c.parallelFor(n / d, @max(1, 16384 / d), Ctx{ .out = c.bf16Of(out), .x = c.bf16Of(out.src[0].?), .w = c.ptrAs(f32, w.data), .d = d, .eps = out.op_params[0] }, rows);
    }
};
src/backend/cpu/norm.zig 修改
// ---- src/backend/cpu/norm.zig: 换掉 RmsNormImpl ----
const NormCtx = struct { out: [*]c.bf16, x: [*]const c.bf16, w: [*]const f32, d: usize, eps: f32 };

fn normRows(ctx: NormCtx, r0: usize, r1: usize) void {
    const d = ctx.d;
    for (r0..r1) |r| {
        const x = ctx.x + r * d;
        var ss: f32 = 0;
        for (0..d) |i| ss += c.bf2f(x[i]) * c.bf2f(x[i]);
        const rstd = 1.0 / @sqrt(ss / @as(f32, @floatFromInt(d)) + ctx.eps);
        const o = ctx.out + r * d;
        // 舍入点和 .cu 一样:(x·rstd) 先舍入到 bf16, 再乘权重, 再舍入
        for (0..d) |i| o[i] = c.f2bf(c.roundBf(c.bf2f(x[i]) * rstd) * ctx.w[i]);
    }
}

/// 行宽 = 权重长度:attn_norm / ffn_norm / final_norm 按 hidden, q_norm / k_norm 按 head_dim
pub const RmsNormImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const x = out_tensor.src[0].?;
        const w = out_tensor.src[1].?;
        if (w.data_type != .GGML_TYPE_F32 or x.data_type != .GGML_TYPE_BF16) return error.RmsNormNeedsBf16AndF32Gain;
        const d = w.numElements();
        c.parallelFor(x.numElements() / d, @max(1, 16384 / d), NormCtx{
            .out = c.bf16Of(out_tensor),
            .x = c.bf16Of(x),
            .w = c.ptrAs(f32, w.data),
            .d = d,
            .eps = out_tensor.op_params[0],
        }, normRows);
    }
};

行宽取权重长度，和 CUDA 的 Impl 同一个约定，所以 q_norm / k_norm 在 token-major 的 [1, T, n, hd] 上不用特殊处理：每 128 个数一行。

原实现里的 w_offset（op_params[1]）在 Qwen3 的图里恒为 0，这里不再读。

fn mk(dtype: T.DataType, data: anytype, dims: [4]i64) Tensor {
    return Tensor{ .data = @ptrCast(data.ptr), .dtype = dtype, .dims = dims, .device = .CPU };

fn toBf(comptime n: usize, v: [n]f32) [n]u16 {
    var o: [n]u16 = undefined;
    for (v, 0..) |x, i| o[i] = bf(x);
    return o;
}

    setStep(.{ .n_tok = 1, .cos_sin = &tab, .pos = &pos });
    try kernel.applyRope(&out, 0);
    try testing.expectEqualSlices(u16, &toBf(2, .{ 0, 1 }), &od);
}

test "gather_last 按 last[] 取行 + sample greedy 取 argmax(并列取小下标)" {
    var xd = toBf(6, .{ 1, 2, 3, 4, 5, 6 }); // [T=3, H=2]
    var x = mk(.GGML_TYPE_BF16, &xd, .{ -1, 2, 0, 0 });
    var od = [_]u16{0} ** 4;

    out.src[0] = &x;

对 llmgen 的预期

Linear 是 CPU 上唯一要紧的算子。早期日志里它平均 87 ms 一次（含 prefill 的大矩阵和 lm_head）。换完之后一个 decode 步的全部 197 次 Linear 加起来应该在 50 ms 左右，对应 lab 的 21 tok/s lab CPU 实测。这大约是两个数量级的差别，来自线程数 × SIMD 宽度 × 去掉逐元素分支。

上限由内存带宽决定：每步读 1.2 GB 权重，双通道 DDR4 约 30–40 GB/s，下限 30–40 ms，也就是 25–30 tok/s 封顶 推算。M 系列芯片的统一内存带宽更高，所以笔记本能跑到 20–29 tok/s。

其余算子在 decode 时合计不到 1 ms；它们的并行化主要帮 prefill。

常见错误

现象原因和处理
Linear 换完后输出全错N 和 K 取反了。llmgen 的权重是 [N, K]：n = dims[0]，k = dims[1]。
多线程比单线程还慢Debug 构建；或者机器上别的进程占着核，自旋的工作线程在抢。用 ReleaseFast。
CPU 和 CUDA 在第 11 步之后开始分叉CPU 的 RMSNorm 没换，舍入点不同。
error.RmsNormNeedsBf16AndF32Gain和 CUDA 同一个约定：norm 权重是 F32。
mac 上 dotBf16 的结果和 x86 差最后一位VLEN 不同（NEON 是 4，AVX2 是 8），部分和的条数不同。f32 层面的正常差别。
prefill 很慢，decode 正常注意力是 O(T²) 的朴素实现（C3）。长 prompt 在 CPU 上本来就慢。

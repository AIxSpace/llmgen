阶段 1 · 去掉主机开销

写 KV 改成一个 kernel

序列级块表、slot 寻址、store_kv kernel。

本步目标写 KV 从“每层 16 次小拷贝”变成“每层一个 kernel”；块表从“每层一份、每步上传”变成“每条序列一份、跟元数据一起上传”。注意力 kernel 换成 lab 的两个（decode 和 prefill），它们的原理放到第 13、14 步讲。做完这步，热路径上只剩 kernel 和一次元数据上传。

原理

起点怎么写 KV

src/memory/page_atten.zig 起点（你现在的代码）
    pub fn appendKVFromTensor(
        self: *Self,
        layer_id: i32,
        k_data: []const u8,
        v_data: []const u8,
        n_kv_heads: i32,
        seq_len: i32,
        head_dim: i32,
        data_tpye: GGMLDataType,
    ) !void {
        const layer_state = self.layers.get(layer_id) orelse return error.LayerNotActive;
        if (!layer_state.active or seq_len <= 0) {
            return;
        }

        const elem_size = data_tpye.size();
        const head_dim_bytes = @as(usize, @intCast(head_dim)) * elem_size;
        const head_stride_bytes = @as(usize, @intCast(seq_len)) * @as(usize, @intCast(head_dim)) * elem_size;

        const prev_cached = layer_state.num_cached;
        try self.appendKVPages(layer_id, seq_len);

        var seq_idx: i32 = 0;
        while (seq_idx < seq_len) : (seq_idx += 1) {
            const global_pos = prev_cached + seq_idx;

            const logical_block: usize = @intCast(@divFloor(global_pos, PAGE_BLOCK_SIZE));
            const offset_in_block: usize = @intCast(@mod(global_pos, PAGE_BLOCK_SIZE));

            const entry = layer_state.page_table.items[logical_block];
            const nk: usize = @intCast(n_kv_heads);
            const kv_base = offset_in_block * nk * @as(usize, @intCast(head_dim)) * elem_size;
            const k_blk = layer_state.k_pool.blockData(entry.k_block_id);
            const v_blk = layer_state.v_pool.blockData(entry.v_block_id);

            var h: usize = 0;
            while (h < nk) : (h += 1) {
                const src_offset = h * head_stride_bytes + @as(usize, @intCast(seq_idx)) * head_dim_bytes; // 输入 head-major
                const dst_offset = kv_base + h * head_dim_bytes; // 块内 token-major:[token][kv_head][hd]

                const device_type = self.pool.getDeviceType(); // pools.zig 里把 getDeviceType 改成 pub
                deviceMemcpy(k_blk[dst_offset .. dst_offset + head_dim_bytes], k_data[src_offset .. src_offset + head_dim_bytes], device_type);
                deviceMemcpy(v_blk[dst_offset .. dst_offset + head_dim_bytes], v_data[src_offset .. src_offset + head_dim_bytes], device_type);
            }
        }
    }

三层循环：token × KV 头 × (K, V)。每次 deviceMemcpy 是一次 cudaMemcpy（D2D），搬 256 字节。decode 一步 28 层共 448 次；512 token 的 prefill 是 512 × 8 × 2 × 28 = 22.9 万次。

        .flash => cu.cuda_flash_attention,

每层在 host 上重新收集一遍 page_table，交给 .cu 去 cudaMalloc、H2D、用完 cudaFree。

两个观察

一、写 KV 是纯拷贝，天然可以并行。每个 token 的每个头是独立的 128 个数。一个 kernel、每个 token 一个 block、每个头一个 warp，一次全搬完。

二、所有层的块表是同一张。第 5 个 token 在第 0 层写进块 0 的第 5 个槽，在第 27 层也是。起点给每层各配了一个 BlockPool 和一张 page_table，28 张表内容完全相同。所以块表应该属于序列，不属于层：

Sequence.block_table逻辑块 → 物理块，host 上，每条序列一份
→
fillMeta算出每个 token 的 slot；块表抄进元数据包
→
store_kv × 28 层读同一份 slots
+
注意力 × 28 层读同一份块表

runner/block.zig 的 BlockManager 和 Sequence.block_table 就是为这个准备的，一直没接上。这一步接上。

slot 和块内布局

slot = 物理块号 × block_size + 块内偏移，一个整数就定位了一个 token 在 KV cache 里的位置。host 上算好放进元数据的 slots[T]，kernel 直接用。

KV cache 每层一大块，布局是 [块][kv_head][块内 slot][head_dim]：

backend/cuda/attention.cu · store_kv lab 参考（3090 实测过）
// ------------------------------------------------------------------
// store_kv: k/v [T, nkv, 128] → 分页 cache 的 slot(块内布局 [kv_head][slot][head_dim]); 一个 warp 一个 KV 头
// ------------------------------------------------------------------
__global__ void store_kv_kernel(const bf16 *k, const bf16 *v, const int *slots, bf16 *kc, bf16 *vc, int nkv,
                                int block_size) {
    const int HD = 128;
    const int t = blockIdx.x, h = threadIdx.x >> 5, lane = threadIdx.x & 31;
    if (h >= nkv) return;
    const int slot = slots[t];
    if (slot < 0) return;
    const long blk = slot / block_size, off = slot % block_size;
    const long dst = ((blk * nkv + h) * block_size + off) * HD;
    const long src = ((long)t * nkv + h) * HD;
#pragma unroll
    for (int j = 0; j < 4; ++j) {
        kc[dst + lane + 32 * j] = k[src + lane + 32 * j];
        vc[dst + lane + 32 * j] = v[src + lane + 32 * j];
    }
}

extern "C" void cuda_store_kv(const cuda_bf16_t *k, const cuda_bf16_t *v, const int *slot_mapping, cuda_bf16_t *k_cache,
                            cuda_bf16_t *v_cache, int T, int nkv, int hd, int block_size) {
    if (T <= 0) return;
    if (hd != 128 || nkv > 32) {
        fprintf(stderr, "[serve] store_kv: 仅支持 head_dim=128 且 nkv<=32\n");
        return;
    }
    store_kv_kernel<<<T, nkv * 32, 0, g_cuda_stream>>>((const bf16 *)k, (const bf16 *)v, slot_mapping, (bf16 *)k_cache,
                                                   (bf16 *)v_cache, nkv, block_size);
}

注意块内是 kv_head 在外、slot 在内，和起点的 [token][kv_head][hd] 相反。原因在读的那一侧：注意力 kernel 一个 block 处理一个 KV 头，它要顺序扫这个头的所有 token。同一个头的 16 个 token 在内存里连续（16 × 256 B = 4 KB），顺序读带宽才上得去。

slot < 0 表示不写，是给第 17 步 CUDA Graph 的 padding 行留的。

src/memory/page_atten.zig（整个文件换掉） 新建文件
//! memory/page_atten.zig -- 每层 K/V 的分页缓存(由 module-serving-unfused 的 paged.zig 改名而来)
//! 块内布局 [kv_head][slot][head_dim]。块的分配 / 回收是序列级的, 由 runner/block.zig 的 BlockManager 管,
//! 所有层共用同一张块表;这里只持有每层 K / V 的显存地址。

const std = @import("std");
const types = @import("../types.zig");
const P = @import("pools.zig");

pub const DataType = types.GGMLDataType;

/// 每块 token 数的缺省值(调度器算池容量时用)
pub const PAGE_BLOCK_SIZE: i32 = 16;

/// 每层的 KV 分页状态(对照 PagedAttentionManager::LayerState).k / v 是设备指针.
pub const LayerState = struct {
    active: bool = false,
    n_kv_heads: i32 = 0,
    head_dim: i32 = 0,
    dtype: DataType = .GGML_TYPE_BF16,
    num_blocks: usize = 0,
    k: ?*anyopaque = null,
    v: ?*anyopaque = null,

    /// 一个 token 的 K(或 V)字节数 = n_kv_heads × head_dim × elem.
    pub fn tokenBytes(self: *const LayerState) usize {
        return @as(usize, @intCast(self.n_kv_heads)) * @as(usize, @intCast(self.head_dim)) * self.dtype.size();
    }
};

/// 对照 class PagedAttentionManager:pool_ + layers_(layer_id → LayerState).
pub const PagedAttentionManager = struct {
    alloc: std.mem.Allocator,
    pool: *P.MemoryPool, // KV 块的 backing(kv_cache 池)
    block_size: usize, // 每块 token 数(module-style 固定 PAGE_BLOCK_SIZE=16, 这里可配)
    layers: std.AutoHashMapUnmanaged(i32, *LayerState) = .empty,

    pub fn init(alloc: std.mem.Allocator, pool: *P.MemoryPool, block_size: usize) PagedAttentionManager {
        return .{ .alloc = alloc, .pool = pool, .block_size = block_size };
    }
    pub fn deinit(self: *PagedAttentionManager) void {
        var it = self.layers.valueIterator();
        while (it.next()) |ls| self.alloc.destroy(ls.*);
        self.layers.deinit(self.alloc);
    }

    pub fn getLayer(self: *PagedAttentionManager, layer_id: i32) ?*LayerState {
        return self.layers.get(layer_id);
    }
    pub fn isActive(self: *PagedAttentionManager, layer_id: i32) bool {
        if (self.layers.get(layer_id)) |ls| return ls.active;
        return false;
    }

    /// 对照 init_layer:只设该层元数据(n_kv/head_dim/dtype),active=true;显存由 reserveLayer 切.
    pub fn initLayer(self: *PagedAttentionManager, layer_id: i32, n_kv_heads: i32, head_dim: i32, dtype: DataType) !void {
        const gop = try self.layers.getOrPut(self.alloc, layer_id);
        if (!gop.found_existing) gop.value_ptr.* = try self.alloc.create(LayerState);
        gop.value_ptr.*.* = .{ .active = true, .n_kv_heads = n_kv_heads, .head_dim = head_dim, .dtype = dtype };
    }

    /// 一层 K(或 V)一块的字节数.
    pub fn blockBytes(self: *const PagedAttentionManager, ls: *const LayerState) usize {
        return self.block_size * ls.tokenBytes();
    }

    /// 对照 reserve_layer:为该层从 KV 池切出 K/V 各 num_blocks 个块.
    pub fn reserveLayer(self: *PagedAttentionManager, layer_id: i32, num_blocks: usize) !void {
        const ls = self.layers.get(layer_id) orelse return error.LayerNotActive;
        if (!ls.active) return;
        const per_pool = self.blockBytes(ls) * num_blocks;
        const kblk = try self.pool.allocate(per_pool, P.POOL_ALIGN);
        const vblk = try self.pool.allocate(per_pool, P.POOL_ALIGN);
        ls.num_blocks = num_blocks;
        ls.k = @ptrCast(kblk.ptr);
        ls.v = @ptrCast(vblk.ptr);
    }
};

//! cuda/attention.zig -- PagedAttentionImpl<CUDA>
//! 先 cuda_store_kv 把本步 k / v 写进本层分页 cache, 再做注意力。
//! decode: split-K flash-decoding(第 13 步讲);prefill: tensor core FlashAttention-2(第 14 步讲)。
const std = @import("std");
const Tensor = @import("../../Tensor.zig");
const c = @import("../cpu/common.zig");
const cu = @import("cuda.zig");

/// decode split-K 的分段长度: 批越小分段越细, 保证 grid = nkv × S × parts 足够铺满 82 个 SM。
pub fn partSize(s: usize) usize {
    return if (s >= 64) 512 else if (s >= 16) 256 else if (s >= 4) 128 else 64;
}
pub fn maxParts(s: usize, max_model_len: usize) usize {
    const ps = partSize(s);
    return (max_model_len + ps - 1) / ps;
}

/// src[0] = q [1, T, nq, hd], src[1] = k [1, T, nkv, hd], src[2] = v [1, T, nkv, hd] → out [1, T, nq, hd]
pub const PagedAttentionImpl = struct {
    pub fn execute(device_id: i32, out: *Tensor) !void {
        _ = device_id;
        const s = &c.g_step;
        const kv = try c.kvOf(out.layer_id);
        cu.cuda_store_kv(out.src[1].?.data, out.src[2].?.data, s.slots, kv.k, kv.v, s.n_tok, s.n_kv_head, s.head_dim, s.block_size);
        if (s.decode) {
            const n_seq: usize = @intCast(s.n_seq);
            cu.cuda_attn_decode(out.data, out.src[0].?.data, kv.k, kv.v, s.block_tables, s.ctx, s.n_seq, s.n_head, s.n_kv_head, s.head_dim, s.max_blocks, s.block_size, s.attn_scale, s.part_o, s.part_ml, @intCast(partSize(n_seq)), @intCast(maxParts(n_seq, s.max_model_len)));
        } else {
            cu.cuda_attn_prefill(out.data, out.src[0].?.data, kv.k, kv.v, s.block_tables, s.ctx, s.cu_q, s.tile_seq, s.tile_q0, s.n_tiles, s.n_head, s.n_kv_head, s.head_dim, s.max_blocks, s.block_size, s.attn_scale);
        }
    }
};

对 llmgen 的预期

这是阶段 1 里收益最大的一步。每步去掉的调用：

  • 448 次 D2D cudaMemcpy（写 KV）+ 56 次 H2D（块表）；
  • 56 次 cudaMalloc + 56 次 cudaFree，按 lab 实测的单次耗时约 4.5 ms 推算；
  • 加上第 05 步去掉的 112 个 permute kernel。

同步的小 cudaMemcpy 单次多贵，lab 没有直接可比的数字（它那张表里的 108.3 µs 搬的是整块权重）。保守按每次 10 µs 算，448 + 56 次是 5 ms。合起来预期 ITL 降 10–20 ms 推算，以你的 nsys 为准。

prefill 的变化更大：512 token 时原来有 22.9 万次小拷贝，现在是 28 个 kernel。S1 的 TTFT 预期从秒级降到百毫秒级 推算（剩下的大头是 lm_head 对 512 行全算，第 08 步解决）。

注意力 kernel 本身也换成了更快的版本，这部分收益在第 13、14 步用回退开关单独量。

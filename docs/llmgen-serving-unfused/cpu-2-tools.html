CPU 支线

CPU：并行与 SIMD 工具

bf16 直读、@Vector 点积、fork-join 线程团队。

本步目标给 CPU 算子准备三样工具：bf16 的直接读写、SIMD 点积、一个 OpenMP 式的线程团队。后面所有 CPU 算子都建在它们上面。建议在主线开始之前就做，否则 CPU 上每验证一步都要等很久。

原理

起点的 CPU 算子慢在哪

src/backend/cpu/linear.zig 起点（你现在的代码）
pub const LinearImpl = struct {
    pub fn execute(_: i32, out_tensor: *Tensor) !void {
        const x = out_tensor.src[0].?;
        const weight = out_tensor.src[1].?;

        const in_dim = common.dimAt(weight, 1);
        const out_dim = common.dimAt(weight, 0);

        const seq_len = common.dimAt(x, 1); // [1, S, H]

        for (0..seq_len) |seq_idx| {
            for (0..out_dim) |out_idx| {
                var acc: f32 = 0;
                const wrow = out_idx * in_dim;
                for (0..in_dim) |in_idx| {
                    acc += common.elemF32(x, seq_idx * in_dim + in_idx) * common.elemF32(weight, wrow + in_idx);
                }
                common.storeF32(out_tensor, seq_idx * out_dim + out_idx, acc);
            }
        }
    }
};
src/backend/cpu/common.zig 起点（你现在的代码）
/// 按 t.data_type 读第 i 个元素为 f32。
pub fn elemF32(t: *const Tensor, i: usize) f32 {
    return loadF32(t.data.?, t.data_type, i);
}

/// 按 dtype 从原始指针读第 i 个元素(给分页 KV 块这类不是 Tensor 的缓冲用)。
pub fn loadF32(base: [*]const u8, data_type: types.GGMLDataType, i: usize) f32 {
    return switch (data_type) {
        .GGML_TYPE_F32 => @as([*]const f32, @ptrCast(@alignCast(base)))[i],
        .GGML_TYPE_BF16 => bf16ToF32(@as([*]const u16, @ptrCast(@alignCast(base)))[i]),
        .GGML_TYPE_F16 => f16ToF32(@as([*]const u16, @ptrCast(@alignCast(base)))[i]),
        else => unsupportedDtype("loadF32", data_type),
    };
}

/// 按 t.data_type 把 f32 写回第 i 个元素
pub fn storeF32(t: *Tensor, i: usize, v: f32) void {
    const tensor_data = t.data.?;
    switch (t.data_type) {
        .GGML_TYPE_F32 => @as([*]f32, @ptrCast(@alignCast(tensor_data)))[i] = v,
        .GGML_TYPE_BF16 => @as([*]u16, @ptrCast(@alignCast(tensor_data)))[i] = f32ToBf16(v),
        .GGML_TYPE_F16 => @as([*]u16, @ptrCast(@alignCast(tensor_data)))[i] = f32ToF16(v),
        else => unsupportedDtype("storeF32", t.data_type),
    }
}

三个问题叠在一起：

  • 单线程。机器有几十个核，只用了一个。
  • 没有 SIMD。一次乘加一个元素。AVX2 一条指令能做 8 个。
  • 每个元素一次 dtype 分支。elemF32 每读一个数都 switch (data_type)。最内层循环里的分支会挡住编译器的自动向量化。

llmgen 早期的 CPU 日志里，一次 Linear 平均 87 ms，占总时间的 99.3% llmgen 早期 CPU 日志。

bf16 直接读

教程全程约定激活和权重都是 bf16。那就不需要通用的 elemF32：把指针当成 [*]u16，读出来左移 16 位就是 f32。没有分支，编译器可以向量化。

SIMD 点积

backend/cpu/common.zig · SIMD（原样加进 llmgen 的 common.zig） lab 参考（3090 实测过）
/// SIMD 通道数: AVX2=8 / NEON=4(同 labs/cpu-parallel).
pub const VLEN: usize = std.simd.suggestVectorLength(f32) orelse 8;
pub const VecF = @Vector(VLEN, f32);

/// 一次读 VLEN 个 bf16 为 f32(零扩展后左移 16 位, 对照 C++ load_bf16x8_f32).
pub inline fn loadBf16(p: [*]const bf16) VecF {
    const bits16: @Vector(VLEN, u16) = p[0..VLEN].*;
    const bits32: @Vector(VLEN, u32) = @intCast(bits16);
    return @bitCast(bits32 << @as(@Vector(VLEN, u5), @splat(16)));
}

/// bf16 点积, f32 累加(n 不必是 VLEN 的倍数).
pub fn dotBf16(a: [*]const bf16, b: [*]const bf16, n: usize) f32 {
    var acc: VecF = @splat(0);
    var i: usize = 0;
    while (i + VLEN <= n) : (i += VLEN) acc = @mulAdd(VecF, loadBf16(a + i), loadBf16(b + i), acc);
    var s = @reduce(.Add, acc);
    while (i < n) : (i += 1) s += bf2f(a[i]) * bf2f(b[i]);
    return s;
}

std.simd.suggestVectorLength(f32) 在 AVX2 上是 8，在 NEON 上是 4，同一份代码两种 CPU 都是满宽度。loadBf16 一次读 VLEN 个 u16，零扩展成 u32 再左移 16 位，按位重解释成 f32 向量。@mulAdd 在支持 FMA 的 CPU 上是一条指令。

累加在 f32 里做，和 .cu 的 GEMV 一样。但求和顺序不同：这里是 VLEN 条部分和最后合并，CUDA 的 GEMV 是 32 个 lane 各自累加再归约。两者的 f32 结果有最后一位的差别；舍入成 bf16 后绝大多数相同。lab 实测端到端 64 个 token 仍然逐个相同 lab CPU 实测。

线程：为什么不用 std.Thread.Pool

直觉的做法是用标准库的线程池：每块任务 spawn 一次。lab 一开始就是这么做的，结果在 32 线程的 Xeon 上 decode 只有 2 tok/s lab CPU 实测。

原因是任务太碎。一个 token 要派发五百多个算子，每个算子切成几十块，每块一次“加锁、入队、条件变量唤醒”。算下来一个 token 要唤醒线程上万次，上下文切换的开销压过了计算本身。

换成 OpenMP 式的常驻 fork-join 线程团队之后是 20.8 tok/s lab CPU 实测，快了 10 倍：

std.Thread.Poolfork-join 团队
派发一次每块：加锁 + 入队 + 唤醒写一次任务描述，代数加一
谁做哪块谁抢到谁做第 i 块固定给第 i 个线程，主线程做第 0 块
空闲时睡在条件变量上先自旋 4096 轮等下一代任务，超时才 futex 睡
小任务照样派发小于 2 × grain 直接在主线程串行做

算子之间的间隙很短（主线程回去派发下一个算子只要几百纳秒），工作线程自旋等着，基本不会睡。这就是 OMP_WAIT_POLICY=active 的效果。

backend/cpu/common.zig · fork-join 线程团队（原样加进 llmgen 的 common.zig） lab 参考（3090 实测过）
// labs/cpu-parallel 用 std.Thread.Pool: 每块一次 spawn(加锁 + 条件变量唤醒). 服务引擎一个 token 要派发
// 几百个算子, 在 32 线程的 Xeon 上这种唤醒开销压过了计算本身(实测 2 tok/s, 每 token 2 万次上下文切换).
// 这里换成 OpenMP 式的常驻 fork-join 线程团队: 第 i 块固定给第 i 个线程(主线程拿第 0 块), 派发 = 写任务 +
// 代数加一; 工作线程先自旋等下一代任务(算子之间的间隙很短, 基本不睡), 自旋超时才 futex 睡眠
// (对照 OMP_WAIT_POLICY / GOMP_SPINCOUNT).
const Team = struct {
    n_workers: usize = 0,
    gen: std.atomic.Value(u32) = .init(0), // 任务代数: 每派发一次加一
    pending: std.atomic.Value(u32) = .init(0), // 本代还没做完的工作线程数
    sleepers: std.atomic.Value(u32) = .init(0), // 正在 futex 上睡的工作线程数
    run_fn: *const fn (*const anyopaque, usize, usize) void = undefined,
    ctx: *const anyopaque = undefined,
    n: usize = 0,
    chunks: usize = 0,

    const SPIN: u32 = 1 << 12;

    fn range(self: *const Team, i: usize) [2]usize {
        const per = (self.n + self.chunks - 1) / self.chunks;
        const b = @min(i * per, self.n);
        return .{ b, @min(b + per, self.n) };
    }

    fn worker(self: *Team, id: usize) void {
        var seen: u32 = 0;
        while (true) {
            var spins: u32 = 0;
            while (self.gen.load(.acquire) == seen) {
                if (spins < SPIN) {
                    spins += 1;
                    std.atomic.spinLoopHint();
                    continue;
                }
                _ = self.sleepers.fetchAdd(1, .seq_cst);
                if (self.gen.load(.seq_cst) == seen) std.Thread.Futex.wait(&self.gen, seen);
                _ = self.sleepers.fetchSub(1, .seq_cst);
            }
            seen = self.gen.load(.acquire);
            if (id < self.chunks) {
                const r = self.range(id);
                if (r[0] < r[1]) self.run_fn(self.ctx, r[0], r[1]);
            }
            _ = self.pending.fetchSub(1, .acq_rel);
        }
    }

    /// 主线程调用: 发布任务 → 自己做第 0 块 → 等所有工作线程做完.
    fn run(self: *Team, n: usize, chunks: usize, ctx: *const anyopaque, f: *const fn (*const anyopaque, usize, usize) void) void {
        self.run_fn = f;
        self.ctx = ctx;
        self.n = n;
        self.chunks = chunks;
        self.pending.store(@intCast(self.n_workers), .release);
        _ = self.gen.fetchAdd(1, .seq_cst);
        if (self.sleepers.load(.seq_cst) > 0) std.Thread.Futex.wake(&self.gen, std.math.maxInt(u32));
        const r = self.range(0);
        f(ctx, r[0], r[1]);
        while (self.pending.load(.acquire) != 0) std.atomic.spinLoopHint();
    }
};

var g_team: Team = .{};
var g_team_once = std.once(initTeam);

fn initTeam() void {
    const n = std.Thread.getCpuCount() catch 1;
    var started: usize = 0;
    for (1..n) |id| {
        const t = std.Thread.spawn(.{ .stack_size = 1 << 20 }, Team.worker, .{ &g_team, id }) catch break;
        t.detach(); // 进程级常驻, 同 OpenMP 的线程团队
        started += 1;
    }
    g_team.n_workers = started;
}

/// 把区间 [0, n) 静态均分成块, 每块调用 body(ctx, start, end); 主线程也参与.
/// grain = 每块至少多少个单位: 块数 = min(线程数, n / grain), 小活直接在主线程串行做.
/// 只从主线程(Executor 派发)调用, 不可嵌套.
pub fn parallelFor(n: usize, grain: usize, ctx: anytype, comptime body: fn (@TypeOf(ctx), usize, usize) void) void {
    if (n == 0) return;
    if (n < 2 * grain) return body(ctx, 0, n);
    g_team_once.call();
    const chunks = @min(g_team.n_workers + 1, n / grain);
    if (chunks <= 1) return body(ctx, 0, n);
    const Ctx = @TypeOf(ctx);
    const Tramp = struct {
        fn call(p: *const anyopaque, b: usize, e: usize) void {
            const cp: *const Ctx = @ptrCast(@alignCast(p));
            body(cp.*, b, e);
        }
    };
    g_team.run(n, chunks, @ptrCast(&ctx), Tramp.call);
}

grain 参数很重要：decode 时一次 add 只有 1024 个元素，开 32 个线程去分反而慢。每个算子按自己的工作量给一个合适的 grain，小活自动退化成串行。

parallelFor 的用法 示意，不落地
// 用法:把 [0, n) 切成块, 每块在一个线程里跑 body(ctx, begin, end)
const Ctx = struct { out: [*]c.bf16, a: [*]const c.bf16, b: [*]const c.bf16 };

fn body(ctx: Ctx, begin: usize, end: usize) void {
    for (begin..end) |i| ctx.out[i] = c.f2bf(c.bf2f(ctx.a[i]) + c.bf2f(ctx.b[i]));
}

// grain = 32768:少于 65536 个元素就不开线程, 直接在主线程做
c.parallelFor(n, 32768, Ctx{ .out = o, .a = a, .b = b }, body);

常见错误

现象原因和处理
parallelFor 之后结果偶尔错body 里写了块之间共享的状态。每块只能写自己那段输出；临时缓冲要放在 body 的栈上（见 C3 注意力的 acc）。
程序退出时卡住工作线程是 detach 的、常驻的，不需要 join。卡住多半是别处在等。
CPU 占用一直 100%（空闲时）自旋等待的代价。SPIN 调小，线程会更早去睡。
incorrect alignment 在 ptrAs张量的 data 没按元素大小对齐。池的分配对齐是 32 / 64，不该出现；多半是传了一个带奇数偏移的指针。
嵌套调用 parallelFor 死锁它不可重入：只能从主线程（Executor 的派发）调用，body 里不能再调。
